93 subroutine time_step_controller_init(this, params)
94 class(time_step_controller_t), intent(inout) :: this
95 type(json_file), intent(inout) :: params
96 integer :: flags_any(2), flags_all(2), ierr
97
98 this%dt_last_change = -1
99 call json_get_or_default(params, 'variable_timestep', &
100 this%is_variable_dt, .false.)
101 if (this%is_variable_dt) then
102 call json_get_or_lookup_or_default(params, 'target_cfl', &
103 this%cfl_trg, 0.4_dp)
104 call json_get_or_lookup_or_default(params, 'timestep', &
105 this%init_dt, huge(0.0_dp))
106 call json_get_or_lookup_or_default(params, 'max_timestep', &
107 this%max_dt, huge(0.0_dp))
108 call json_get_or_lookup_or_default(params, 'min_timestep', &
109 this%min_dt, 0.0_dp)
110 call json_get_or_lookup_or_default(params, 'max_update_frequency',&
111 this%max_update_frequency, 0)
112 call json_get_or_lookup_or_default(params, 'min_update_frequency',&
113 this%min_update_frequency, huge(0))
114 call json_get_or_lookup_or_default(params, 'cfl_running_avg_coeff', &
115 this%alpha, 0.5_dp)
116 call json_get_or_lookup_or_default(params, 'max_dt_increase_factor', &
117 this%max_dt_increase_factor, 1.2_dp)
118 call json_get_or_lookup_or_default(params, 'min_dt_decrease_factor', &
119 this%min_dt_decrease_factor, 0.5_dp)
120 call json_get_or_lookup_or_default(params, 'cfl_deviation_tolerance', &
121 this%dev_tol, 0.2_dp)
122 end if
123
124 call json_get_or_default(params, 'exact_output_time', &
125 this%exact_output_time, .false.)
126 if (this%exact_output_time .and. .not. this%is_variable_dt) then
127 call neko_error('exact_output_time requires variable_timestep')
128 end if
129
130 ! A variable time step takes a collective at every step in an MPMD run,
131 ! and the landing on the scheduled times another one, so the coupled
132 ! simulations have to agree on both, or their collectives would not
133 ! match up.
134 if (is_mpmd()) then
135 flags_any = merge(1, 0, [this%is_variable_dt, this%exact_output_time])
136 flags_all = flags_any
137 call mpi_allreduce(mpi_in_place, flags_any, 2, mpi_integer, &
138 mpi_max, neko_global_comm, ierr)
139 call mpi_allreduce(mpi_in_place, flags_all, 2, mpi_integer, &
140 mpi_min, neko_global_comm, ierr)
141 if (any(flags_any .ne. flags_all)) then
142 call neko_error('variable_timestep and exact_output_time must &
143 &each be set in all the coupled cases of an MPMD run, or in &
144 &none')
145 end if
146 end if
147
148 end subroutine time_step_controller_init

177 subroutine time_step_controller_set_dt(this, time, cfl)
178 class(time_step_controller_t), intent(inout) :: this
179 type(time_state_t), intent(inout) :: time
180 real(kind=dp), intent(in) :: cfl
181 real(kind=dp) :: dt_old, scaling_factor, global_min_dt
182 character(len=LOG_SIZE) :: log_buf
183 integer :: ierr
184
185 this%dt_previous = time%dt
186
187 ! Check if variable dt is requested
188 if (.not. this%is_variable_dt) return
189
190 ! Reset the average cfl if it is the first time step since the last change
191 if (this%dt_last_change .eq. 0) then
192 this%cfl_avg = cfl
193 end if
194
195 if (this%dt_last_change .eq. -1) then
196
197 ! Set the first dt for desired cfl, or use the provided initial dt if it
198 ! is smaller. Then clamp between max and min dt if provided.
199 time%dt = this%first_dt(time, cfl)
200 this%dt_last_change = 0
201 this%cfl_avg = cfl
202 ! There is no step before the first one
203 this%dt_previous = time%dt
204
205 else
206 ! Calculate the average of cfl over the desired interval
207 this%cfl_avg = this%alpha * cfl + (1 - this%alpha) * this%cfl_avg
208
209 if (abs(this%cfl_avg - this%cfl_trg) .ge. this%dev_tol * this%cfl_trg &
210 .and. this%dt_last_change .ge. this%max_update_frequency &
211 .or. this%dt_last_change .ge. this%min_update_frequency) then
212
213 if (this%cfl_trg/cfl .ge. 1) then
214 ! increase of time step
215 scaling_factor = min(this%max_dt_increase_factor, this%cfl_trg/cfl)
216 else
217 ! reduction of time step
218 scaling_factor = max(this%min_dt_decrease_factor, this%cfl_trg/cfl)
219 end if
220
221 dt_old = time%dt
222 time%dt = scaling_factor * dt_old
223 time%dt = max(min(time%dt, this%max_dt), this%min_dt)
224
225 write(log_buf, '(A,E15.7,1x,A,E15.7)') &
226 'Average CFL:', this%cfl_avg, &
227 'Target CFL:', this%cfl_trg
228 call neko_log%message(log_buf)
229
230 write(log_buf, '(A,E15.7,1x,A,E15.7)') 'Old dt:', dt_old, &
231 'New dt:', time%dt
232 call neko_log%message(log_buf)
233
234 this%dt_last_change = 0
235
236 else
237 this%dt_last_change = this%dt_last_change + 1
238 end if
239 end if
240
241 ! If running in mpmd, the new dt is the minimum across simulations
242 if (is_mpmd()) then
243 global_min_dt = time%dt
244 call mpi_allreduce(mpi_in_place, global_min_dt, 1, &
245 mpi_double_precision, mpi_min, neko_global_comm, ierr)
246
247 ! If my dt is larger that the global min, mark a change
248 if (time%dt .gt. global_min_dt) then
249 time%dt = global_min_dt
250 this%dt_last_change = 0
251 end if
252 end if
253
254
255 end subroutine time_step_controller_set_dt

278 subroutine time_step_controller_land(this, time, time_to_next)
279 class(time_step_controller_t), intent(inout) :: this
280 type(time_state_t), intent(inout) :: time
281 real(kind=dp), intent(in) :: time_to_next
282 real(kind=dp) :: direction, dt_prev, lo, remaining, remaining_end, x
283 integer(kind=i8) :: n
284 integer :: ierr
285
286 if (.not. (this%exact_output_time .and. this%is_variable_dt)) return
287
288 direction = sign(1.0_dp, time%dt)
289 dt_prev = abs(this%dt_previous)
290 lo = this%min_dt_decrease_factor * dt_prev
291
292 ! The end, or the next scheduled time unless it is too close
293 remaining_end = direction * (time%end_time - time%t)
294 if (remaining_end .le. time_to_next + lo) then
295 remaining = remaining_end
296 else if (time_to_next .ge. lo) then
297 remaining = time_to_next
298 else
299 remaining = 0.0_dp
300 end if
301
302 x = remaining / abs(time%dt)
303 if (remaining .gt. 0.0_dp .and. x .lt. max_landing_steps) then
304 ! Candidate count: x proposed steps fit into the remaining time,
305 ! round down, but take at least one step
306 n = max(1_i8, floor(x, kind = i8))
307 ! Round up instead if x is closer to n + 1 in ratio, that is, if
308 ! (n + 1) / x < x / n. The boundary is the geometric mean
309 ! sqrt(n (n + 1)), not n + 0.5, so that a step twice too long and one
310 ! half as long count as equally far off
311 if (x * x .gt. real(n, dp) * real(n + 1_i8, dp)) n = n + 1_i8
312 ! Lower bound on the step, min_dt_decrease_factor times the previous
313 ! one: at most remaining / lo steps, and at least one. The tolerance
314 ! keeps round-off from dropping a count that fits exactly
315 n = min(n, max(1_i8, floor(remaining / lo + landing_tol, kind = i8)))
316 ! Upper bound on the step, max_dt_increase_factor times the previous
317 ! one or max_timestep, whichever is smaller: at least that many
318 ! steps. Applied last, so it wins if the two bounds conflict
319 n = max(n, ceiling(remaining / min(this%max_dt_increase_factor * &
320 dt_prev, this%max_dt) - landing_tol, kind = i8))
321 ! The fitted step: n equal steps end exactly on the target
322 time%dt = direction * remaining / real(n, dp)
323 end if
324
325 ! In an MPMD run the simulations advance in lockstep
326 if (is_mpmd()) then
327 call mpi_allreduce(mpi_in_place, time%dt, 1, mpi_double_precision, &
328 mpi_min, neko_global_comm, ierr)
329 end if
330
331 ! The projection spaces watch for a change of the step taken
332 if (abs(abs(time%dt) - dt_prev) .gt. landing_tol * dt_prev) then
333 this%dt_last_change = 0
334 end if
335
336 end subroutine time_step_controller_land
