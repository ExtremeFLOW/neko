49static void dynamic_smagorinsky_jit() {
50 if (dynamic_smagorinsky_nut_program == NULL) {
51 opencl_kernel_jit(dynamic_smagorinsky_nut_kernel,
52 (cl_program *) &dynamic_smagorinsky_nut_program);
53 }

57static void dynamic_smagorinsky_run(cl_kernel kernel, const int n) {
58 if (n > 0) {
59 const int nb = (n + 256 - 1) / 256;
60 const size_t global_item_size = 256 * nb;
61 const size_t local_item_size = 256;
62
63 CL_CHECK(clEnqueueNDRangeKernel((cl_command_queue) glb_cmd_queue, kernel,
64 1, NULL, &global_item_size,
65 &local_item_size, 0, NULL, NULL));
66 }
67
68 CL_CHECK(clReleaseKernel(kernel));
69}
