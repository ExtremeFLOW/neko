676 * Reduction buffer
677 */
678 int red_s = 0;
679 real * bufred = NULL;
680 void * bufred_d = NULL;
681
685 void cuda_redbuf_check_alloc(int nb) {
686 if ( nb >= red_s) {
687 red_s = nb+1;
688 if (bufred != NULL) {
689 CUDA_CHECK(cudaFreeHost(bufred));
690#ifdef HAVE_NVSHMEM
691 nvshmem_free(bufred_d);
692#else
693 CUDA_CHECK(cudaFree(bufred_d));
694#endif
695 }
696 CUDA_CHECK(cudaMallocHost(&bufred,red_s*sizeof(real)));
697#ifdef HAVE_NVSHMEM
698 bufred_d = (real *) nvshmem_malloc(red_s*sizeof(real));
699#else
700 CUDA_CHECK(cudaMalloc(&bufred_d, red_s*sizeof(real)));
701#endif
702 }
703 }

708 void cuda_global_reduce_add(real * bufred, void * bufred_d,
709 int n, const cudaStream_t stream) {
710
711#ifdef HAVE_NCCL
712 device_nccl_allreduce(bufred_d, bufred_d, n, sizeof(real),
713 DEVICE_NCCL_SUM, stream);
714 CUDA_CHECK(cudaMemcpyAsync(bufred, bufred_d, sizeof(real)*n,
715 cudaMemcpyDeviceToHost, stream));
716 cudaStreamSynchronize(stream);
717#elif HAVE_NVSHMEM
718 if (sizeof(real) == sizeof(float)) {
719 nvshmemx_float_sum_reduce_on_stream(NVSHMEM_TEAM_WORLD,
720 (float *) bufred_d,
721 (float *) bufred_d, n, stream);
722 }
723 else if (sizeof(real) == sizeof(double)) {
724 nvshmemx_double_sum_reduce_on_stream(NVSHMEM_TEAM_WORLD,
725 (double *) bufred_d,
726 (double *) bufred_d, n, stream);
727
728 }
729 CUDA_CHECK(cudaMemcpyAsync(bufred, bufred_d,
730 sizeof(real)*n, cudaMemcpyDeviceToHost, stream));
731 cudaStreamSynchronize(stream);
732#elif HAVE_DEVICE_MPI
733 cudaStreamSynchronize(stream);
734 device_mpi_allreduce(bufred_d, bufred, n, sizeof(real), DEVICE_MPI_SUM);
735#else
736 CUDA_CHECK(cudaMemcpyAsync(bufred, bufred_d, n*sizeof(real),
737 cudaMemcpyDeviceToHost, stream));
738 cudaStreamSynchronize(stream);
739#endif
740 }

745 void cuda_global_reduce_max(real * bufred, void * bufred_d,
746 int n, const cudaStream_t stream) {
747
748#ifdef HAVE_NCCL
