728 void cuda_global_reduce_add(real * bufred, void * bufred_d,
729 int n, const cudaStream_t stream) {
730
731#ifdef HAVE_NCCL
732 device_nccl_allreduce(bufred_d, bufred_d, n, sizeof(real),
733 DEVICE_NCCL_SUM, stream);
734 CUDA_CHECK(cudaMemcpyAsync(bufred, bufred_d, sizeof(real)*n,
735 cudaMemcpyDeviceToHost, stream));
736 cudaStreamSynchronize(stream);
737#elif HAVE_NVSHMEM
738 if (sizeof(real) == sizeof(float)) {
739 nvshmemx_float_sum_reduce_on_stream(NVSHMEM_TEAM_WORLD,
740 (float *) bufred_d,
741 (float *) bufred_d, n, stream);
742 }
743 else if (sizeof(real) == sizeof(double)) {
744 nvshmemx_double_sum_reduce_on_stream(NVSHMEM_TEAM_WORLD,
745 (double *) bufred_d,
746 (double *) bufred_d, n, stream);
747
748 }
749 CUDA_CHECK(cudaMemcpyAsync(bufred, bufred_d,
750 sizeof(real)*n, cudaMemcpyDeviceToHost, stream));
