192 function bicgstab_cpld_solve(this, Ax, x, y, z, fx, fy, fz, n, coef, &
193 bc_projector, gs_h, niter) result(ksp_results)
194 class(bicgstab_cpld_t), intent(inout) :: this
195 class(ax_t), intent(in) :: ax
196 type(field_t), intent(inout) :: x
197 type(field_t), intent(inout) :: y
198 type(field_t), intent(inout) :: z
199 integer, intent(in) :: n
200 real(kind=rp), dimension(n), intent(in) :: fx
201 real(kind=rp), dimension(n), intent(in) :: fy
202 real(kind=rp), dimension(n), intent(in) :: fz
203 type(coef_t), intent(inout) :: coef
204 class(vector_bc_projector_t), intent(inout) :: bc_projector
205 type(gs_t), intent(inout) :: gs_h
206 integer, optional, intent(in) :: niter
207 type(ksp_monitor_t), dimension(3) :: ksp_results
208 integer :: i, iter, max_iter, ierr
209 real(kind=rp) :: alpha, beta, omega, rho_1, rho_2
210 real(kind=rp) :: rnorm, norm_fac, gamma
211 real(kind=rp) :: r_norm, s_norm, shadow_norm, t_norm, v_norm
212 ! r^T r, s^T s, f^T v, v^T v, s^T t, t^T t
213 real(kind=rp) :: rtr, sts, ftv, vtv, stt, ttt
214 real(kind=xp) :: norm_sum
215 type(host_array_t), pointer :: p_tmp, p_hat_tmp, r_tmp
216 type(host_array_t), pointer :: s_hat_tmp, t_tmp, v_tmp
217 integer :: temp_indices(6)
218
219 if (present(niter)) then
220 max_iter = niter
221 else
222 max_iter = this%max_iter
223 end if
224 norm_fac = 1.0_rp / sqrt(coef%volume)
225
226 call neko_scratch_registry%request_host_array(p_tmp, temp_indices(1), &
227 3 * n, .false.)
228 call neko_scratch_registry%request_host_array(p_hat_tmp, &
229 temp_indices(2), 3 * n, .false.)
230 call neko_scratch_registry%request_host_array(r_tmp, temp_indices(3), &
231 3 * n, .false.)
232 call neko_scratch_registry%request_host_array(s_hat_tmp, &
233 temp_indices(4), 3 * n, .false.)
234 call neko_scratch_registry%request_host_array(t_tmp, temp_indices(5), &
235 3 * n, .false.)
236 call neko_scratch_registry%request_host_array(v_tmp, temp_indices(6), &
237 3 * n, .false.)
238
239 this%p(1:n, 1:3) => p_tmp%x
240 this%p_hat(1:n, 1:3) => p_hat_tmp%x
241 this%r(1:n, 1:3) => r_tmp%x
242 this%s_hat(1:n, 1:3) => s_hat_tmp%x
243 this%t(1:n, 1:3) => t_tmp%x
244 this%v(1:n, 1:3) => v_tmp%x
245
246 associate(p => this%p, p_hat => this%p_hat, r => this%r, &
247 s_hat => this%s_hat, t => this%t, v => this%v)
248
249 ! BiCGStab starts from zero. The right-hand side is consequently both
250 ! the initial residual and the fixed shadow residual.
251 norm_sum = 0.0_xp
252 !$omp parallel do reduction(+:norm_sum)
253 do i = 1, n
254 x%x(i, 1, 1, 1) = 0.0_rp
255 y%x(i, 1, 1, 1) = 0.0_rp
256 z%x(i, 1, 1, 1) = 0.0_rp
257 r(i, 1) = fx(i)
258 r(i, 2) = fy(i)
259 r(i, 3) = fz(i)
260 norm_sum = norm_sum + coef%mult(i, 1, 1, 1) * &
261 (r(i, 1)**2 + r(i, 2)**2 + r(i, 3)**2)
262 end do
263 !$omp end parallel do
264 call mpi_allreduce(mpi_in_place, norm_sum, 1, mpi_extra_precision, &
265 mpi_sum, neko_comm, ierr)
266
267 rtr = norm_sum
268 r_norm = bicgstab_cpld_sqrt(rtr, 'initial residual')
269 shadow_norm = r_norm
270 rnorm = r_norm * norm_fac
271 gamma = rnorm * this%rel_tol
272 ksp_results%res_start = rnorm
273 ksp_results%res_final = rnorm
274 ksp_results%iter = 0
275
276 ! Besides saving an iteration, the early check prevents a converged
277 ! right-hand side from being reported as a rho breakdown.
278 if (r_norm .le. 0.0_rp .or. rnorm .lt. this%abs_tol .or. &
279 rnorm .lt. gamma) then
280 ksp_results%converged = .true.
281 nullify(this%p, this%p_hat, this%r, this%s_hat, this%t, this%v)
282 call neko_scratch_registry%relinquish_host_array(temp_indices)
283 return
284 end if
285
286 call this%monitor_start('Coupled BiCGStab')
287 do iter = 1, max_iter
288 call bicgstab_cpld_product(rho_1, fx, fy, fz, r(:, 1), r(:, 2), &
289 r(:, 3), coef%mult, n)
290
291 ! The combined product-space norms make the breakdown decision
292 ! invariant under a uniform scaling of the complete vector system.
293 call bicgstab_cpld_check_inner_product(rho_1, shadow_norm, r_norm, &
294 'rho inner product')
295
296 if (iter .eq. 1) then
297 !$omp parallel do
298 do i = 1, n
299 p(i, 1) = r(i, 1)
300 p(i, 2) = r(i, 2)
301 p(i, 3) = r(i, 3)
302 end do
303 !$omp end parallel do
304 else
305 beta = (rho_1 / rho_2) * (alpha / omega)
306 if (.not. ieee_is_finite(beta)) then
307 call neko_error('Coupled BiCGStab failure: non-finite beta')
308 end if
309
310 !$omp parallel do
311 do i = 1, n
312 p(i, 1) = r(i, 1) + &
313 beta * (p(i, 1) - omega * v(i, 1))
314 p(i, 2) = r(i, 2) + &
315 beta * (p(i, 2) - omega * v(i, 2))
316 p(i, 3) = r(i, 3) + &
317 beta * (p(i, 3) - omega * v(i, 3))
318 end do
319 !$omp end parallel do
320 end if
321
322 ! The preconditioner interface is scalar, so apply the same
323 ! preconditioner separately to all three components.
324 call this%M%solve(p_hat(:, 1), p(:, 1), n)
325 call this%M%solve(p_hat(:, 2), p(:, 2), n)
326 call this%M%solve(p_hat(:, 3), p(:, 3), n)
327
328 call ax%compute_vector(v(:, 1), v(:, 2), v(:, 3), p_hat(:, 1), &
329 p_hat(:, 2), p_hat(:, 3), coef, x%msh, x%Xh)
330 call bicgstab_cpld_assemble(v, n, coef, bc_projector, gs_h)
331
332 ! Compute f^T v and v^T v in a single global reduction. The latter
333 ! supplies the scale for the alpha-denominator breakdown check.
334 call bicgstab_cpld_product_and_norm(ftv, vtv, fx, fy, fz, &
335 v(:, 1), v(:, 2), v(:, 3), coef%mult, n)
336 v_norm = bicgstab_cpld_sqrt(vtv, 'operator result v')
337 call bicgstab_cpld_check_inner_product(ftv, shadow_norm, v_norm, &
338 'alpha denominator')
339 alpha = rho_1 / ftv
340 if (.not. ieee_is_finite(alpha)) then
341 call neko_error('Coupled BiCGStab failure: non-finite alpha')
342 end if
343
344 ! Store the intermediate residual in r and compute its combined norm
345 ! in the same pass. The previous residual is no longer needed after
346 ! p has been formed.
347 norm_sum = 0.0_xp
348 !$omp parallel do reduction(+:norm_sum)
349 do i = 1, n
350 r(i, 1) = r(i, 1) - alpha * v(i, 1)
351 r(i, 2) = r(i, 2) - alpha * v(i, 2)
352 r(i, 3) = r(i, 3) - alpha * v(i, 3)
353 norm_sum = norm_sum + coef%mult(i, 1, 1, 1) * &
354 (r(i, 1)**2 + r(i, 2)**2 + r(i, 3)**2)
355 end do
356 !$omp end parallel do
357 call mpi_allreduce(mpi_in_place, norm_sum, 1, mpi_extra_precision, &
358 mpi_sum, neko_comm, ierr)
359
360 sts = norm_sum
361 s_norm = bicgstab_cpld_sqrt(sts, 'intermediate residual')
362 rnorm = s_norm * norm_fac
363 if (rnorm .lt. this%abs_tol .or. rnorm .lt. gamma) then
364 !$omp parallel do
365 do i = 1, n
366 x%x(i, 1, 1, 1) = x%x(i, 1, 1, 1) + alpha * p_hat(i, 1)
367 y%x(i, 1, 1, 1) = y%x(i, 1, 1, 1) + alpha * p_hat(i, 2)
368 z%x(i, 1, 1, 1) = z%x(i, 1, 1, 1) + alpha * p_hat(i, 3)
369 end do
370 !$omp end parallel do
371 call this%monitor_iter(iter, rnorm)
372 exit
373 end if
374
375 call this%M%solve(s_hat(:, 1), r(:, 1), n)
376 call this%M%solve(s_hat(:, 2), r(:, 2), n)
377 call this%M%solve(s_hat(:, 3), r(:, 3), n)
378
379 call ax%compute_vector(t(:, 1), t(:, 2), t(:, 3), s_hat(:, 1), &
380 s_hat(:, 2), s_hat(:, 3), coef, x%msh, x%Xh)
381 call bicgstab_cpld_assemble(t, n, coef, bc_projector, gs_h)
382
383 ! The numerator and denominator of omega share one reduction.
384 call bicgstab_cpld_product_and_norm(stt, ttt, r(:, 1), r(:, 2), &
385 r(:, 3), t(:, 1), t(:, 2), t(:, 3), coef%mult, n)
386 t_norm = bicgstab_cpld_sqrt(ttt, 'operator result t')
387 if (t_norm .le. 0.0_rp) then
388 call neko_error(&
389 'Coupled BiCGStab breakdown: zero omega denominator')
390 end if
391 if (.not. ieee_is_finite(stt)) then
392 call neko_error(&
393 'Coupled BiCGStab failure: non-finite omega numerator')
394 end if
395 omega = stt / ttt
396 if (.not. ieee_is_finite(omega)) then
397 call neko_error('Coupled BiCGStab failure: non-finite omega')
398 end if
399
400 ! Update the solution, recursive residual, and residual norm together
401 ! to avoid another traversal of all three component vectors.
402 norm_sum = 0.0_xp
403 !$omp parallel do reduction(+:norm_sum)
404 do i = 1, n
405 x%x(i, 1, 1, 1) = x%x(i, 1, 1, 1) + alpha * p_hat(i, 1) + &
406 omega * s_hat(i, 1)
407 y%x(i, 1, 1, 1) = y%x(i, 1, 1, 1) + alpha * p_hat(i, 2) + &
408 omega * s_hat(i, 2)
409 z%x(i, 1, 1, 1) = z%x(i, 1, 1, 1) + alpha * p_hat(i, 3) + &
410 omega * s_hat(i, 3)
411 r(i, 1) = r(i, 1) - omega * t(i, 1)
412 r(i, 2) = r(i, 2) - omega * t(i, 2)
413 r(i, 3) = r(i, 3) - omega * t(i, 3)
414 norm_sum = norm_sum + coef%mult(i, 1, 1, 1) * &
415 (r(i, 1)**2 + r(i, 2)**2 + r(i, 3)**2)
416 end do
417 !$omp end parallel do
418 call mpi_allreduce(mpi_in_place, norm_sum, 1, mpi_extra_precision, &
419 mpi_sum, neko_comm, ierr)
420
421 rtr = norm_sum
422 r_norm = bicgstab_cpld_sqrt(rtr, 'recursive residual')
423 rnorm = r_norm * norm_fac
424 call this%monitor_iter(iter, rnorm)
425 if (rnorm .lt. this%abs_tol .or. rnorm .lt. gamma) then
426 exit
427 end if
428
429 ! A negative omega is valid. Only numerical orthogonality between s
430 ! and t constitutes a breakdown of the minimal-residual step.
431 call bicgstab_cpld_check_inner_product(stt, s_norm, t_norm, &
432 'omega numerator')
433 rho_2 = rho_1
434 end do
435
436 call this%monitor_stop()
437 ksp_results%res_final = rnorm
438 ksp_results%iter = iter
439 ksp_results%converged = this%is_converged(iter, rnorm)
440 end associate
441 nullify(this%p, this%p_hat, this%r, this%s_hat, this%t, this%v)
442 call neko_scratch_registry%relinquish_host_array(temp_indices)
443
444 end function bicgstab_cpld_solve

451 subroutine bicgstab_cpld_check_inner_product(inner_product, norm_a, norm_b, &
452 quantity)
453 real(kind=rp), intent(in) :: inner_product
454 real(kind=rp), intent(in) :: norm_a
455 real(kind=rp), intent(in) :: norm_b
456 character(len=*), intent(in) :: quantity
457 real(kind=rp) :: large_norm, small_norm
458
459 if (.not. ieee_is_finite(inner_product)) then
460 call neko_error('Coupled BiCGStab failure: non-finite ' // &
461 trim(quantity))
462 end if
463
464 ! Divide by the larger norm first to avoid overflow in their product.
465 large_norm = max(norm_a, norm_b)
466 small_norm = min(norm_a, norm_b)
467 if (large_norm .le. 0.0_rp .or. &
468 abs(inner_product) / large_norm .le. neko_eps * small_norm) then
469 call neko_error('Coupled BiCGStab breakdown: near-zero ' // &
470 trim(quantity))
471 end if
472
473 end subroutine bicgstab_cpld_check_inner_product

498 subroutine bicgstab_cpld_assemble(vector, n, coef, bc_projector, gs_h)
499 integer, intent(in) :: n
500 real(kind=rp), dimension(n, 3), intent(inout) :: vector
501 type(coef_t), intent(inout) :: coef
502 class(vector_bc_projector_t), intent(inout) :: bc_projector
503 type(gs_t), intent(inout) :: gs_h
504
505 ! Cyclic faces must be expressed in their common coordinate system while
506 ! the three components are gathered, then rotated back to physical space.
507 call rotate_cyc(vector(:, 1), vector(:, 2), vector(:, 3), 1, coef)
508 call gs_h%op(vector(:, 1), vector(:, 2), vector(:, 3), n, gs_op_add)
509 call rotate_cyc(vector(:, 1), vector(:, 2), vector(:, 3), 0, coef)
510
511 call bc_projector%apply(vector(:, 1), vector(:, 2), vector(:, 3), n)
512
513 end subroutine bicgstab_cpld_assemble

525 subroutine bicgstab_cpld_product(product, ax, ay, az, bx, by, bz, mult, n)
526 integer, intent(in) :: n
527 real(kind=rp), intent(out) :: product
528 real(kind=rp), dimension(n), intent(in) :: ax, ay, az
529 real(kind=rp), dimension(n), intent(in) :: bx, by, bz
530 real(kind=rp), dimension(n), intent(in) :: mult

532 integer :: i, ierr
533
534 product_sum = 0.0_xp
535 !$omp parallel do reduction(+:product_sum)
536 do i = 1, n
537 product_sum = product_sum + mult(i) * &

539 end do
540 !$omp end parallel do
541
542 call mpi_allreduce(mpi_in_place, product_sum, 1, mpi_extra_precision, &
543 mpi_sum, neko_comm, ierr)

559 subroutine bicgstab_cpld_product_and_norm(product, norm_squared, ax, ay, &
560 az, bx, by, bz, mult, n)
561 integer, intent(in) :: n
562 real(kind=rp), intent(out) :: product
563 real(kind=rp), intent(out) :: norm_squared
564 real(kind=rp), dimension(n), intent(in) :: ax, ay, az
565 real(kind=rp), dimension(n), intent(in) :: bx, by, bz
566 real(kind=rp), dimension(n), intent(in) :: mult
567 real(kind=xp) :: product_sum, norm_sum
568 real(kind=xp) :: reductions(2)
569 integer :: i, ierr
570
571 product_sum = 0.0_xp
572 norm_sum = 0.0_xp
573 !$omp parallel do reduction(+:product_sum,norm_sum)
574 do i = 1, n
575 product_sum = product_sum + mult(i) * &
576 (ax(i) * bx(i) + ay(i) * by(i) + az(i) * bz(i))
577 norm_sum = norm_sum + mult(i) * &
578 (bx(i)**2 + by(i)**2 + bz(i)**2)
579 end do
580 !$omp end parallel do
581
582 reductions(1) = product_sum
583 reductions(2) = norm_sum
584 call mpi_allreduce(mpi_in_place, reductions, 2, mpi_extra_precision, &
585 mpi_sum, neko_comm, ierr)
586 product = reductions(1)
587 norm_squared = reductions(2)
588
589 end subroutine bicgstab_cpld_product_and_norm
