349 subroutine scalar_scheme_init(this, msh, c_Xh, gs_Xh, params, scheme, user, &
350 rho)
351 class(scalar_scheme_t), target, intent(inout) :: this
352 type(mesh_t), target, intent(in) :: msh
353 type(coef_t), target, intent(in) :: c_Xh
354 type(gs_t), target, intent(inout) :: gs_Xh
355 type(json_file), target, intent(inout) :: params
356 character(len=*), intent(in) :: scheme
357 type(user_t), target, intent(in) :: user
358 type(field_t), target, intent(in) :: rho
359 ! IO buffer for log output
360 character(len=LOG_SIZE) :: log_buf
361 ! Variables for retrieving json parameters
362 logical :: logical_val
363 real(kind=rp) :: real_val, solver_abstol
364 integer :: integer_val, ierr
365 character(len=:), allocatable :: solver_type, solver_precon
366 type(json_file) :: precon_params
367 type(json_file) :: json_subdict
368 logical :: nut_dependency
369
370 this%u => neko_registry%get_field('u')
371 this%v => neko_registry%get_field('v')
372 this%w => neko_registry%get_field('w')
373 this%rho => rho
374
375 ! Assign a name
376 ! Note that the keyword is added by `scalars_t`, so there is always a
377 ! default.
378 call json_get(params, 'name', this%name)
379
380 ! Set the freeze flag
381 call json_get_or_default(params, 'freeze', this%freeze, .false.)
382
383 call neko_log%section('Scalar')
384 call json_get(params, 'solver.type', solver_type)
385 call json_get(params, 'solver.preconditioner.type', &
386 solver_precon)
387 call json_get(params, 'solver.preconditioner', precon_params)
388 call json_get_or_lookup(params, 'solver.absolute_tolerance', &
389 solver_abstol)
390
391 call json_get_or_lookup_or_default(params, &
392 'solver.projection_space_size', &
393 this%projection_dim, 0)
394 call json_get_or_lookup_or_default(params, &
395 'solver.projection_hold_steps', &
396 this%projection_activ_step, 5)
397
398
399 write(log_buf, '(A, A)') 'Type : ', trim(scheme)
400 call neko_log%message(log_buf)
401 write(log_buf, '(A, A)') 'Name : ', trim(this%name)
402 call neko_log%message(log_buf)
403 call neko_log%message('Ksp scalar : ('// trim(solver_type) // &
404 ', ' // trim(solver_precon) // ')')
405 write(log_buf, '(A,ES13.6)') ' `-abs tol :', solver_abstol
406 call neko_log%message(log_buf)
407
408 this%Xh => this%u%Xh
409 this%dm_Xh => this%u%dof
410 if (associated(this%params)) then
411 deallocate(this%params)
412 end if
413 allocate(this%params)
414 this%params = params
415 this%msh => msh
416
417 call neko_registry%add_field(this%dm_Xh, this%name, &
418 ignore_existing = .true.)
419
420 this%s => neko_registry%get_field(this%name)
421
422 call this%slag%init(this%s, 2)
423
424 this%gs_Xh => gs_xh
425 this%c_Xh => c_xh
426
427 !
428 ! Material properties
429 !
430 call this%set_material_properties(params, user)
431
432 !
433 ! Spectral vanishing viscosity
434 !
435 if (params%valid_path('svv')) then
436 call json_get_or_default(params, 'svv.enabled', this%svv_enabled, &
437 .false.)
438 if (this%svv_enabled) then
439 allocate(this%svv)
440 call this%svv%init(params, this%c_Xh, this%rho)
441 end if
442 end if
443
444
445 !
446 ! Turbulence modelling
447 !
448 this%alphat_field_name = ""
449 this%nut_field_name = ""
450 if (params%valid_path('alphat')) then
451 call json_get(this%params, 'alphat', json_subdict)
452 call json_get(json_subdict, 'nut_dependency', nut_dependency)
453 if (nut_dependency) then
454 call json_get_or_lookup(json_subdict, 'Pr_t', this%pr_turb)
455 call json_get(json_subdict, 'nut_field', this%nut_field_name)
456 else
457 call json_get(json_subdict, 'alphat_field', this%alphat_field_name)
458 end if
459 end if
460
461 !
462 ! Setup right-hand side field.
463 !
464 allocate(this%f_Xh)
465 call this%f_Xh%init(this%dm_Xh, fld_name = "scalar_rhs")
466
467 ! Initialize the source term
468 call this%source_term%init(this%f_Xh, this%c_Xh, user, this%name)
469 call this%source_term%add(params, 'source_terms')
470
471 ! todo parameter file ksp tol should be added
472 call json_get_or_lookup_or_default(params, &
473 'solver.max_iterations', &
474 integer_val, ksp_max_iter)
475 call json_get_or_default(params, &
476 'solver.monitor', &
477 logical_val, .false.)
478 call scalar_scheme_solver_factory(this%ksp, this%dm_Xh%size(), &
479 solver_type, integer_val, solver_abstol, logical_val)
480 call scalar_scheme_precon_factory(this%pc, this%ksp, &
481 this%c_Xh, this%dm_Xh, this%gs_Xh, this%bcs, &
482 solver_precon, precon_params)
483
484 call neko_log%end_section()
485
486 end subroutine scalar_scheme_init

531 subroutine scalar_scheme_free(this)
532 class(scalar_scheme_t), intent(inout) :: this
533 class(bc_t), pointer :: bc
534 integer :: i
535
536 bc => null()
537
538 if (allocated(this%svv)) then
539 call this%svv%free()
540 deallocate(this%svv)
541 end if
542 this%svv_enabled = .false.
543
544 nullify(this%Xh)
545 nullify(this%dm_Xh)
546 nullify(this%gs_Xh)
547 nullify(this%c_Xh)
548 if (associated(this%params)) then
549 deallocate(this%params)
550 nullify(this%params)
551 end if
552
553 if (allocated(this%ksp)) then
554 call this%ksp%free()
555 deallocate(this%ksp)
556 end if
557
558 if (allocated(this%pc)) then
559 call precon_destroy(this%pc)
560 deallocate(this%pc)
561 end if
562
563 if (allocated(this%name)) then
564 deallocate(this%name)
565 end if
566
567 call this%source_term%free()
568
569 if (associated(this%f_Xh)) then
570 call this%f_Xh%free()
571 deallocate(this%f_Xh)
572 end if
573
574 do i = 1, this%bcs%size()
575 bc => this%bcs%get(i)
576 if (associated(bc)) then
577 call bc%free()
578 deallocate(bc)
579 end if
580 end do
581
582 call this%bcs%free()
583 call this%slag%free()
584 call this%material_properties%free()
585
586 if (allocated(this%nut_field_name)) then
587 deallocate(this%nut_field_name)
588 end if
589
590 if (allocated(this%alphat_field_name)) then
591 deallocate(this%alphat_field_name)
592 end if
593
594 nullify(this%cp)
595 nullify(this%lambda)
596 nullify(this%lambda_tot)
597 nullify(bc)
598
599 end subroutine scalar_scheme_free

692 subroutine scalar_scheme_update_material_properties(this, time)
693 class(scalar_scheme_t), intent(inout) :: this
694 type(time_state_t), intent(in) :: time
695 type(field_t), pointer :: nut, alphat
696 integer :: index
697 ! Factor to transform nu_t to lambda_t
698 type(field_t), pointer :: lambda_factor
699
700 call this%user_material_properties(this%name, this%material_properties, &
701 time)
702
703 ! factor = rho * cp / pr_turb
704 if (len_trim(this%nut_field_name) .gt. 0 &
705 .and. len_trim(this%alphat_field_name) .eq. 0 ) then
706 nut => neko_registry%get_field(this%nut_field_name)
707
708 ! lambda_tot = lambda + rho * cp * nut / pr_turb
709 call neko_scratch_registry%request_field(lambda_factor, index, .false.)
710 call field_cmult2(lambda_factor, nut, 1.0_rp / this%pr_turb)
711 call field_col2(lambda_factor, this%cp)
712 call field_col2(lambda_factor, this%rho)
713 call field_add3(this%lambda_tot, this%lambda, lambda_factor)
714 call neko_scratch_registry%relinquish_field(index)
715
716 else if (len_trim(this%alphat_field_name) .gt. 0 &
717 .and. len_trim(this%nut_field_name) .eq. 0 ) then
718 alphat => neko_registry%get_field(this%alphat_field_name)
719
720 ! lambda_tot = lambda + rho * cp * alphat
721 call neko_scratch_registry%request_field(lambda_factor, index, .false.)
722 call field_col3(lambda_factor, this%cp, alphat)
723 call field_col2(lambda_factor, this%rho)
724 call field_add3(this%lambda_tot, this%lambda, lambda_factor)
725 call neko_scratch_registry%relinquish_field(index)
726
727 else if (len_trim(this%alphat_field_name) .gt. 0 &
728 .and. len_trim(this%nut_field_name) .gt. 0 ) then
729 call neko_error("Conflicting definition of eddy diffusivity " // &
730 "for the scalar equation")
731 end if
732

747 subroutine scalar_scheme_set_material_properties(this, params, user)
748 class(scalar_scheme_t), intent(inout) :: this
749 type(json_file), intent(inout) :: params
750 type(user_t), target, intent(in) :: user
751 character(len=LOG_SIZE) :: log_buf
752 ! A local pointer that is needed to make Intel happy
753 procedure(user_material_properties_intf), pointer :: dummy_mp_ptr
754 real(kind=rp) :: const_cp, const_lambda
755 ! Dummy time state set to 0
756 type(time_state_t) :: time
757
758 dummy_mp_ptr => dummy_user_material_properties
759
760 ! Fill lambda field with the physical value
761
762 call neko_registry%add_field(this%dm_Xh, this%name // "_lambda")
763 call neko_registry%add_field(this%dm_Xh, this%name // "_lambda_tot")
764 call neko_registry%add_field(this%dm_Xh, this%name // "_cp")
765 this%lambda => neko_registry%get_field(this%name // "_lambda")
766 this%lambda_tot => neko_registry%get_field(this%name // "_lambda_tot")
767 this%cp => neko_registry%get_field(this%name // "_cp")
768
769 call this%material_properties%init(2)
770 call this%material_properties%assign(1, this%cp)
771 call this%material_properties%assign(2, this%lambda)
772
773 if (.not. associated(user%material_properties, dummy_mp_ptr)) then
774
775 write(log_buf, '(A)') "Material properties must be set in the user " // &
776 "file!"
777 call neko_log%message(log_buf)
778 this%user_material_properties => user%material_properties
779
780 call user%material_properties(this%name, this%material_properties, time)
781 else
782 this%user_material_properties => dummy_user_material_properties
783 if (params%valid_path('Pe') .and. &
784 (params%valid_path('lambda') .or. &
785 params%valid_path('cp'))) then
786 call neko_error("To set the material properties for the scalar, " // &
787 "either provide Pe OR lambda and cp in the case file.")
788 ! Non-dimensional case
789 else if (params%valid_path('Pe')) then
790 write(log_buf, '(A)') 'Non-dimensional scalar material properties' //&
791 ' input.'
792 call neko_log%message(log_buf, lvl = neko_log_verbose)
793 write(log_buf, '(A)') 'Specific heat capacity will be set to 1, '
794 call neko_log%message(log_buf, lvl = neko_log_verbose)
795 write(log_buf, '(A)') 'conductivity to 1/Pe. Assumes density is 1.'
796 call neko_log%message(log_buf, lvl = neko_log_verbose)
797
798 ! Read Pe into lambda for further manipulation.
799 call json_get_or_lookup(params, 'Pe', const_lambda)
800 write(log_buf, '(A,ES13.6)') 'Pe :', const_lambda
801 call neko_log%message(log_buf)
802
803 ! Set cp and rho to 1 since the setup is non-dimensional.
804 const_cp = 1.0_rp
805 ! Invert the Pe to get conductivity
806 const_lambda = 1.0_rp/const_lambda
807 ! Dimensional case
808 else
809 call json_get_or_lookup(params, 'lambda', const_lambda)
810 call json_get_or_lookup(params, 'cp', const_cp)
811 end if
812 end if
813 ! We need to fill the fields based on the parsed const values
814 ! if the user routine is not used.
815 if (associated(user%material_properties, dummy_mp_ptr)) then
816 ! Fill mu and rho field with the physical value
817 call field_cfill(this%lambda, const_lambda)
818 call field_cfill(this%cp, const_cp)
819
820 write(log_buf, '(A,ES13.6)') 'lambda :', const_lambda
821 call neko_log%message(log_buf)
822 write(log_buf, '(A,ES13.6)') 'cp :', const_cp
823 call neko_log%message(log_buf)
824 end if
825
826 ! Copy over material property to the total one
827 call field_copy(this%lambda_tot, this%lambda)
828
829 ! Since cp is a field and we use the %x(1,1,1,1) of the
830 ! host array data to pass constant material properties
831 ! to some routines, we need to make sure that the host
832 ! values are also filled
833 if (neko_bcknd_device .eq. 1) then
834 call device_memcpy(this%cp%x, this%cp%x_d, this%cp%size(), &
835 device_to_host, sync = .false.)
836 end if
837 end subroutine scalar_scheme_set_material_properties
