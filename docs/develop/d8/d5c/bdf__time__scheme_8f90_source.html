69 !> Implicit backward-differencing scheme for time integration.
70 !! @details
71 !! The explicit forumlas for the coefficients are taken from the following
72 !! techincal note:
73 !! "Derivation of BDF2/BDF3 for Variable Step Size" by Hiroaki Nishikawa,
74 !! which can be found on ResearchGate.
75 !!
76 !! For a contant time-step this corresponds to the following schemes for
77 !! order 1 to 3:
78 !! - Order 1: \f$ \frac{1}{\Delta t} u^{n+1} - \frac{1}{\Delta t} u^{n} \f$
79 !! - Order 2: \f$ \frac{3}{2\Delta t} u^{n+1} - \frac{4}{2\Delta t} u^{n} +
80 !! \frac{1}{2\Delta t} u^{n-1}\f$
81 !! - Order 3: \f$ \frac{11}{6\Delta t} u^{n+1} - \frac{18}{6\Delta t} u^{n} +
82 !! \frac{9}{6\Delta t} u^{n-1} - \frac{2}{6\Delta t} u^{n-2}\f$
83 !!
84 !! It is assumed that all the coefficients but the first one premultiply terms
85 !! that go to the right-hand side of the equation.
86 !! Accordingly, the signs of these coefficients are reversed in the `coeffs`
87 !! array. This is taken into account, for example, in the implemeation of the
88 !! `rhs_maker` class.
89 !!
90 !! Another important convention is that the coefficients are meant to be
91 !! later divided by the current value of the timestep.
92 !!
93 !! In line with the above assumptions, the first order scheme always returns
94 !! the array \f$[1, 1]\f$, and **not** \f$[1/\Delta t, -1/\Delta t]\f$, as one
95 !! might expect. Similar for the second and third order.
96 !!
97 !! @remark
98 !! The current implementation can be easily extended to schemes of arbitrary
99 !! order, by using the `fd_weights_full` subroutine to compute the
100 !! coefficients. A demonstration of this is implemented in a test in
101 !! `tests/ext_bdf_scheme/test_bdf.pf`
102 type, public, extends(time_scheme_t) :: bdf_time_scheme_t
103 contains
104 !> Compute the scheme coefficients
105 procedure, nopass :: compute_coeffs => bdf_time_scheme_compute_coeffs
106 end type bdf_time_scheme_t
107
108contains

110 !> Compute the scheme coefficients
111 !! @param t Timestep values, first element is the current timestep.
112 !! @param order Order the scheme.
113 subroutine bdf_time_scheme_compute_coeffs(coeffs, dt, order)
114 implicit none
115 real(kind=rp), intent(out) :: coeffs(4)
116 real(kind=rp), intent(in) :: dt(10)
117 integer, intent(in) :: order
118
119 call rzero(coeffs, 4)
120
121 ! Note, these are true coeffs, multiplied by dt(1)
122 select case (order)
123 case (1)
124 coeffs(1) = 1.0_rp
125 coeffs(2) = 1.0_rp
126 case (2)
127 coeffs(1) = (1 + dt(1)/(dt(1) + dt(2)))
128 coeffs(3) = -dt(1)**2/dt(2)/(dt(1) + dt(2))
129 coeffs(2) = coeffs(1) - coeffs(3)
130 case (3)
131 coeffs(2) = (dt(1) + dt(2)) * (dt(1) + dt(2) + dt(3)) / &
132 (dt(1) * dt(2) * (dt(2) + dt(3)))
133 coeffs(3) = -dt(1) * (dt(1) + dt(2) + dt(3)) / &
134 (dt(2) * dt(3) * (dt(1) + dt(2)))
135 coeffs(4) = dt(1) * (dt(1) + dt(2)) / &
136 (dt(3) * (dt(2) + dt(3)) * (dt(1) + dt(2) + dt(3)))
137 coeffs(1) = coeffs(2) + coeffs(3) + coeffs(4)
138 coeffs = coeffs * dt(1)
139 case default
140 call neko_error("The order of the BDF time scheme must be 1 to 3.")
141 end select
142
143 end subroutine bdf_time_scheme_compute_coeffs
