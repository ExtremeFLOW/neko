96/*
97 * Reports whether the device code really was compiled for a matrix core
98 * architecture, i.e. whether the __gfx90a__ / __gfx942__ guard below was true
99 * in the device pass.
100 *
101 * This is not the same question as "does the device have matrix cores". The
102 * contraction primitives and their call sites are all guarded on those
103 * macros, so a build whose offload arch does not include the running device's
104 * -- or a code object selected from a fat binary built for something else --
105 * turns the whole strategy into a silent no-op: the kernel launches, writes
106 * nothing, and leaves stale values in the output. That fails as bad results
107 * rather than as an error, which is the worst way for it to fail. Checking it
108 * from the device removes the guesswork.
109 */
110__global__ void hip_mfma_arch_probe(int * flag) {
111#if defined(__gfx90a__) || defined(__gfx942__)
112 *flag = 1;
113#else
114 *flag = 0;
115#endif
116}

126static inline bool hip_have_mfma() {
127 static int cached = -1;
128 if (cached < 0) {
129 int dev = 0;
130 hipDeviceProp_t prop;
131 cached = 0;
132 if (hipGetDevice(&dev) == hipSuccess &&
133 hipGetDeviceProperties(&prop, dev) == hipSuccess &&
134 (strstr(prop.gcnArchName, "gfx90a") != NULL ||
135 strstr(prop.gcnArchName, "gfx942") != NULL)) {
136 int *d_flag = NULL;
137 int flag = 0;
138 if (hipMalloc(&d_flag, sizeof(int)) == hipSuccess) {
139 if (hipMemcpy(d_flag, &flag, sizeof(int),
140 hipMemcpyHostToDevice) == hipSuccess) {
141 hipLaunchKernelGGL(hip_mfma_arch_probe, dim3(1), dim3(1), 0, 0,
142 d_flag);
143 if (hipGetLastError() == hipSuccess &&
144 hipMemcpy(&flag, d_flag, sizeof(int),
145 hipMemcpyDeviceToHost) == hipSuccess) {
146 cached = flag;
147 }
148 }
149 hipFree(d_flag);
150 }
151 }
152 }
153 return cached == 1;
154}

174/*
175 * Wavefronts per block for the MFMA kernels. Candidate C selects 2^C
176 * wavefronts, i.e. 1, 2, 4 or 8.
177 *
178 * Two things are being traded. The contraction stripes its N column groups
179 * across wavefronts, and there are only NGROUPS = ceil(LX^2/16) of them --
180 * one at LX = 4, four at LX = 8 -- so past that count the extra wavefronts
181 * idle through the matrix core work. The staging and pointwise loops, on the
182 * other hand, keep scaling: at LX = 8, eight wavefronts is 512 threads for
183 * 512 points, one each, which is why the branch this came from defaulted to
184 * eight. The measured LX = 8 curve was still improving at four, hence the
185 * fourth candidate.
186 */
187#define NEKO_MFMA_CANDIDATES 4
188#define NEKO_MFMA_NWF(C) (1 << (C))
189#define NEKO_MFMA_NTHRDS(C) dim3(64, NEKO_MFMA_NWF(C), 1)
190
191/*
192 * Column groups per contraction -- the wavefront-parallel work one element
193 * offers. Both tiles group N the same way, 16 columns at a time, so this is
194 * ceil(LX^2/16) either way: 1 at LX = 4, 4 at LX = 8, 9 at LX = 12.
195 *
196 * A wavefront beyond that count has no matrix core work left on that element,
197 * which is what the LX = 4 single precision sweep measured: 20.5 / 23.6 /
198 * 34.0 / 60.2 us as NWF went 1, 2, 4, 8, monotonically worse, against
199 * 218 / 157 / 136.4 / 136.4 at LX = 8 where four groups exist. Rather than
200 * cap the sweep, the surplus wavefronts are given their own element: NWF is
201 * read as wavefronts per block, WPE = min(NWF, NGROUPS) of them cooperate on
202 * one element, and the block covers EB = NWF/WPE elements. At LX = 4 with
203 * eight wavefronts that is eight elements, one each, with nothing idle; at
204 * LX = 12 it is one element and eight cooperating wavefronts, exactly as
205 * before. This matters because p-multigrid smooths at LX = 4 and 2, so low
206 * order Ax is hot rather than incidental.
207 */
208#define NEKO_MFMA_NGROUPS(LX) (((LX) * (LX) + 15) / 16)
209/* Wavefronts cooperating on one element */
210#define NEKO_MFMA_WPE(LX, C) \
211 (NEKO_MFMA_NWF(C) < NEKO_MFMA_NGROUPS(LX) ? NEKO_MFMA_NWF(C) \
212 : NEKO_MFMA_NGROUPS(LX))
213/* Elements per block */
214#define NEKO_MFMA_EB(LX, C) (NEKO_MFMA_NWF(C) / NEKO_MFMA_WPE(LX, C))
215#define NEKO_MFMA_NBLCKS(NELV, LX, C) \
216 dim3(((NELV) + NEKO_MFMA_EB(LX, C) - 1) / NEKO_MFMA_EB(LX, C), 1, 1)
217
218/*
219 * Whether the autotuner sweeps the MFMA strategy, on by default wherever the
220 * hardware and the polynomial order allow it.
221 *
222 * It was briefly off while the matrix core contraction was known broken -- the
223 * lane layout had the block selector and the contraction index interchanged,
224 * which a "the solver converges" check had failed to catch for a long time.
225 * The layout was measured on gfx90a, corrected, and mfma_contract_4x4 now
226 * reproduces a CPU reference to ~1e-16 over every supported order, axis,
227 * transpose/accumulate mode and wavefront count, so there is no reason to
228 * withhold it from the sweep. Kept as an off switch in the shape of
229 * NEKO_EB_TUNE, for A/B work.
230 */

271/*

277struct mfma_traits;

301/*
302 * Linearised index into an LX^3 cube stored as i + LX*j + LX*LX*k, where the
303 * coordinate 'p' lies on contraction axis AXIS and 'n' enumerates the two
304 * remaining axes as n = a + LX*b.
305 */
306template< const int LX, const int AXIS >
307__device__ __forceinline__ int mfma_cube_idx(const int p, const int n) {
308 const int a = n % LX;
309 const int b = n / LX;
310 if (AXIS == 0) return p + LX * a + LX * LX * b; // contract i; n = (j,k)
311 if (AXIS == 1) return a + LX * p + LX * LX * b; // contract j; n = (i,k)
312 return a + LX * b + LX * LX * p; // contract k; n = (i,j)
313}

315/*
316 * One wavefront contracts the reference derivative matrix 'dmat' (LX x LX,
317 * stored column-major: D(row,col) = dmat[row + LX*col]) with the cube 'in'
318 * along axis AXIS, writing the cube 'out':
319 *
320 * out[idx(m,n)] (+)= sum_l D(m,l) * in[idx(l,n)] (TRANSPOSE = false)
321 * out[idx(m,n)] (+)= sum_l D(l,m) * in[idx(l,n)] (TRANSPOSE = true)
322 *
323 * GEMM dimensions M = LX, N = LX*LX, K = LX, tiled over 16x16x4 MFMA tiles
324 * with the partial M/N/K tiles masked. ACCUM selects += over = .
325 */
326/*
327 * NWF cooperating wavefronts (wf = 0..NWF-1) stripe the NTILES N-tiles among
328 * themselves -- wavefront wf handles nt = wf, wf+NWF, ... NWF = 1, wf = 0
329 * (the defaults) reproduces the single-wavefront contraction.
330 */
331template< typename T, const int LX, const int AXIS,
332 const bool TRANSPOSE, const bool ACCUM, const int NWF = 1 >
333__device__ __forceinline__
334void mfma_contract(T * __restrict__ out,
335 const T * __restrict__ dmat,
336 const T * __restrict__ in,
337 const int lane, const int wf = 0) {
338 typedef mfma_traits<T> mma_t;
339 const int g = lane >> 4; // lane / 16 -> 0..3
340 const int c = lane & 15; // lane % 16 -> 0..15
341 const int NTILES = (LX * LX + 15) / 16;
342 const int KSTEPS = (LX + 3) / 4;
343 const int NPASS = (NTILES + NWF - 1) / NWF; // N-tiles handled by this wave
344
345#pragma unroll
346 for (int p = 0; p < NPASS; ++p) {
347 const int nt = wf + p * NWF; // this wavefront's N-tile
348 if (nt < NTILES) {
349 const int n = nt * 16 + c; // free (column) index
350 typename mma_t::acc_t acc = {0, 0, 0, 0};
351#pragma unroll
352 for (int ks = 0; ks < KSTEPS; ++ks) {
353 const int l = ks * 4 + g; // contraction index
354 T a = 0;
355 if (c < LX && l < LX)
356 a = TRANSPOSE ? dmat[l + c * LX] // D(l,c) = D^T(c,l)
357 : dmat[c + l * LX]; // D(c,l)
358 T b = 0;
359 if (l < LX && n < LX * LX)
360 b = in[mfma_cube_idx<LX, AXIS>(l, n)];
361 acc = mma_t::mma(a, b, acc);
362 }
363 const T dvals[4] = { acc[0], acc[1], acc[2], acc[3] };
364#pragma unroll
365 for (int r = 0; r < 4; ++r) {
366 const int m = mma_t::out_row(g, r); // output coordinate on AXIS
367 if (m < LX && n < LX * LX) {
368 const int idx = mfma_cube_idx<LX, AXIS>(m, n);
369 if (ACCUM) out[idx] += dvals[r];
370 else out[idx] = dvals[r];
371 }
372 }
373 }
374 }
375}

377/*
378 * Double-precision batched matrix-core contraction using v_mfma_f64_4x4x4f64.
379 *
380 * Same contract as mfma_contract() -- out(m,n) (+)= sum_l D(m,l) in(l,n), with
381 * D(l,m) when TRANSPOSE -- but tiled with 4x4x4 MFMA tiles and the
382 * instruction's four blocks assigned to four consecutive 4-column N-subtiles.
383 * GEMM M = LX, N = LX*LX, K = LX, tiled as MT = ceil(LX/4) M-tiles,
384 * KSTEPS = ceil(LX/4) K-steps and NGROUPS = ceil(LX^2/16) column groups (each
385 * group = 4 blocks x 4 columns); partial M/N/K masked with zeros.
386 *
387 * Versus the 16x16x4 tile, the 4-wide M granularity fills M = LX < 16 exactly
388 * (LX = 8 runs the matrix core at 100% M-utilisation instead of 50%), at the
389 * cost of 4x as many, 1/4-sized MFMA issues that feed the same f64 matrix
390 * pipeline. Double precision only: the f32 4x4 instruction has K = 1, so there
391 * is no single-precision counterpart -- single precision keeps the 16x16x4 tile
392 * (see mfma_contract_sel below).
393 *
394 * Wave of 64 lanes = 4 blocks x 16 lanes. v_mfma_f64_4x4x4f64 lane layout,
395 * with lo = lane%4, gemm = (lane/4)%4 and kq = lane/16:
396 * A[i][k] : lane holds A[i = lo ][k = kq]
397 * B[k][j] : lane holds B[k = kq ][j = lo]
398 * D[i][j] : lane holds D[i = kq ][j = lo] (scalar f64 accumulator)
399 * and the four independent 4x4x4 blocks are selected by 'gemm', not by
400 * lane/16.
401 *
402 * This was MEASURED on gfx90a, not assumed: a one-hot read-out of which lane
403 * receives which element (128 launches, no assumptions) produced exactly this
404 * mapping. The previous version had 'gemm' and 'kq' interchanged -- it used
405 * lane/16 as the block selector and (lane/4)%4 as the contraction index -- and
406 * every one of 72 probe configurations disagreed with a CPU reference.
407 *
408 * It had been believed validated because the Ax-helm fluid solver converged
409 * with it. It does: a wrong but symmetric operator makes CG converge happily
410 * to the solution of a different system. Convergence is not correctness, and
411 * only a diff against a reference settles a layout. Any future correction
412 * stays localised to the four index expressions (m_a, l, n, m_d).
413 *
414 * NWF cooperating wavefronts (wf = 0..NWF-1) stripe the NGROUPS column groups
415 * among themselves -- wavefront wf handles ng = wf, wf+NWF, ... NWF = 1,
416 * wf = 0 (the defaults) reproduces the single-wavefront contraction.
417 */
418template< const int LX, const int AXIS,
419 const bool TRANSPOSE, const bool ACCUM, const int NWF = 1 >
420__device__ __forceinline__
421void mfma_contract_4x4(double * __restrict__ out,
422 const double * __restrict__ dmat,
423 const double * __restrict__ in,
424 const int lane, const int wf = 0) {
425 const int lo = lane & 3; // 0..3 : A row, B column, D column
426 const int gemm = (lane >> 2) & 3; // 0..3 : which of the four 4x4x4 blocks
427 const int kq = lane >> 4; // 0..3 : contraction index within a step
428 const int MT = (LX + 3) / 4;
429 const int KSTEPS = (LX + 3) / 4;
430 const int NGROUPS = (LX * LX + 15) / 16;
431 const int NPASS = (NGROUPS + NWF - 1) / NWF; // groups handled by this wave
432
433#pragma unroll
434 for (int mt = 0; mt < MT; ++mt) {
435 const int m_a = mt * 4 + lo; // A row (input lane layout: i = lo)
436 const int m_d = mt * 4 + kq; // D row (output lane layout: i = kq)
437#pragma unroll
438 for (int gp = 0; gp < NPASS; ++gp) {
439 const int ng = wf + gp * NWF; // this wavefront's column group
440 if (ng < NGROUPS) {
441 /* the four blocks take four consecutive 4-column N-subtiles */
442 const int n = ng * 16 + gemm * 4 + lo; // column (N) for B in / D out
443 double acc = 0.0;
444#pragma unroll
445 for (int ks = 0; ks < KSTEPS; ++ks) {
446 const int l = ks * 4 + kq; // contraction index (k = kq for A and B)
447 double a = 0.0;
448 if (m_a < LX && l < LX)
449 a = TRANSPOSE ? dmat[l + m_a * LX] // D(l,m) = D^T(m,l)
450 : dmat[m_a + l * LX]; // D(m,l)
451 double b = 0.0;
452 if (l < LX && n < LX * LX)
453 b = in[mfma_cube_idx<LX, AXIS>(l, n)];
454 acc = __builtin_amdgcn_mfma_f64_4x4x4f64(a, b, acc, 0, 0, 0);
455 }
456 if (m_d < LX && n < LX * LX) {
457 const int idx = mfma_cube_idx<LX, AXIS>(m_d, n);
458 if (ACCUM) out[idx] += acc;
459 else out[idx] = acc;
460 }
461 }
462 }
463 }
464}

466/*
467 * Precision-dispatched tensor contraction: double precision uses the batched
468 * 4x4x4 matrix core (full M-utilisation for M = LX < 16), single precision the
469 * 16x16x4 tile (no f32 4x4x4 equivalent). Lets one call site cover both.

476 const T * __restrict__ in, const int lane,
477 const int wf = 0) {
478 mfma_contract<T, LX, AXIS, TRANSPOSE, ACCUM, NWF>(out, dmat, in, lane, wf);
479 }
