67void opencl_bicgstab_update_p(void *p, void *r, void *v, real *beta,
68 real *omega, int *n,
69 cl_command_queue cmd_queue) {
70 cl_int err;
71
72 if (*n <= 0) {
73 return;
74 }
75
76 if (bicgstab_program == NULL)
77 opencl_kernel_jit(bicgstab_kernel, (cl_program *) &bicgstab_program);
78
79 cl_kernel kernel = clCreateKernel(bicgstab_program,
80 "bicgstab_update_p_kernel", &err);
81 CL_CHECK(err);
82
83 CL_CHECK(clSetKernelArg(kernel, 0, sizeof(cl_mem), (void *) &p));
84 CL_CHECK(clSetKernelArg(kernel, 1, sizeof(cl_mem), (void *) &r));
85 CL_CHECK(clSetKernelArg(kernel, 2, sizeof(cl_mem), (void *) &v));
86 CL_CHECK(clSetKernelArg(kernel, 3, sizeof(real), beta));
87 CL_CHECK(clSetKernelArg(kernel, 4, sizeof(real), omega));
88 CL_CHECK(clSetKernelArg(kernel, 5, sizeof(int), n));
89
90 const int nb = ((*n) + 256 - 1) / 256;
91 const size_t global_item_size = 256 * nb;
92 const size_t local_item_size = 256;
93
94 CL_CHECK(clEnqueueNDRangeKernel(cmd_queue, kernel, 1, NULL,
95 &global_item_size, &local_item_size,
96 0, NULL, NULL));
97 CL_CHECK(clReleaseKernel(kernel));
98}

104void opencl_bicgstab_product_and_norm(void *a, void *b, void *mult,
105 real_xp *res, int *n,
106 cl_command_queue cmd_queue) {
107 cl_int err;
108 cl_event kern_wait;
109 int i;
110
111 res[0] = 0.0;
112 res[1] = 0.0;
113 if (*n <= 0) {
114 return;
115 }
116
117 if (bicgstab_program == NULL)
118 opencl_kernel_jit(bicgstab_kernel, (cl_program *) &bicgstab_program);
119
120 const int nb = ((*n) + 256 - 1) / 256;
121 const size_t global_item_size = 256 * nb;
122 const size_t local_item_size = 256;
123
124 opencl_buffer_reserve(&redbuf, 2 * nb * sizeof(real_xp));
125
126 cl_kernel kernel = clCreateKernel(bicgstab_program,
127 "bicgstab_product_and_norm_kernel",
128 &err);
129 CL_CHECK(err);
130
131 CL_CHECK(clSetKernelArg(kernel, 0, sizeof(cl_mem), (void *) &a));
132 CL_CHECK(clSetKernelArg(kernel, 1, sizeof(cl_mem), (void *) &b));
133 CL_CHECK(clSetKernelArg(kernel, 2, sizeof(cl_mem), (void *) &mult));
134 CL_CHECK(clSetKernelArg(kernel, 3, sizeof(cl_mem), (void *) &redbuf.dev));
135 CL_CHECK(clSetKernelArg(kernel, 4, sizeof(int), n));
136 CL_CHECK(clSetKernelArg(kernel, 5, sizeof(int), &nb));
137
138 CL_CHECK(clEnqueueNDRangeKernel(cmd_queue, kernel, 1, NULL,
139 &global_item_size, &local_item_size,
140 0, NULL, &kern_wait));
141
142 CL_CHECK(clEnqueueReadBuffer(cmd_queue, redbuf.dev, CL_TRUE, 0,
143 2 * nb * sizeof(real_xp),
144 ((real_xp *) redbuf.host), 1,
145 &kern_wait, NULL));
146
147 for (i = 0; i < nb; i++) {
148 res[0] += ((real_xp *) redbuf.host)[i];
149 res[1] += ((real_xp *) redbuf.host)[nb + i];
150 }
151
152 CL_CHECK(clReleaseEvent(kern_wait));
153 CL_CHECK(clReleaseKernel(kernel));
154}
