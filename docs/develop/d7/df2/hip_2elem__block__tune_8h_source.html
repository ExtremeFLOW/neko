66/* Sweep the elements per block candidates?
67 On by default, as on CUDA. It was off here because the blocked variants
68 measured a loss on MI250X and MI300A, where they spill -- but baking that
69 in also stopped the tuner from ever re-testing it, and the kstep family has

154/*
155 * Loop over the candidates of one formulation, binding C to each in turn.
156 *
157 * ON gates the formulation itself: hardware support, and -- when NEKO_AUTOTUNE
158 * is set -- whether it is the formulation that was asked for. PIN is the
159 * candidate that formulation's own variable forces, or -1 to measure all N of
160 * them. So NEKO_AUTOTUNE narrows the search to one kernel family without
161 * deciding the geometry within it, which is still swept and reported.
162 *
163 * An out of play formulation yields an empty range, leaving its candidates at
164 * NEKO_TUNE_INIT so it loses the comparison at the end rather than having to
165 * be excluded from it.
166 */
167#define NEKO_TUNE_FOR(C, ON, PIN, N) \
168 for (int C = (((PIN) >= 0) ? (PIN) : 0), \
169 C##_last_ = ((ON) ? (((PIN) >= 0) ? (PIN) + 1 : (N)) : 0); \
170 C < C##_last_; C++)
171
172/* One timed round of LAUNCH at candidate C, min reduced into T[C] */
173#define NEKO_TUNE_TIME(T, LAUNCH, LX, C, ITERS) \
174 do { \
175 float t_; \
176 HIP_CHECK(hipEventRecord(start, stream)); \
177 for (int i_ = 0; i_ < (ITERS); i_++) { LAUNCH(LX, C); } \
178 HIP_CHECK(hipEventRecord(stop, stream)); \
179 HIP_CHECK(hipEventSynchronize(stop)); \
180 HIP_CHECK(hipEventElapsedTime(&t_, start, stop)); \
181 if (t_ < (T)[C]) { (T)[C] = t_; } \
182 } while (0)
183
184/*
185 * Elapsed time (ms, over ITERS launches) as microseconds per call.
186 *
187 * A division rather than a constant because ITERS is NEKO_TUNE_ITERS, which
188 * is settable: the factor of 10 this used to carry is only correct at the
189 * default of 100 and silently rescales every reported time otherwise. The
190 * ranking never moves, every candidate sharing the divisor, but the number
191 * the log prints does.
192 */
193#define NEKO_TUNE_US(T, ITERS) ((T) * 1000.0 / (double) (ITERS))
194
195#define NEKO_TUNE_BEST(T, BEST, N) \
196 do { \
197 for (int c = 1; c < (N); c++) { \
198 if ((T)[c] < (T)[BEST]) { BEST = c; } \
199 } \
200 } while (0)

202/* Report every measured candidate of one sweep, not just the winner. Split
203 per formulation because the vector operator has no 1d variant and so wants
204 the kstep half on its own */
205#define NEKO_TUNE_LOG_1D(LX, T1) \
206 do { \
207 for (int c = 0; c < NEKO_CHUNKS_CANDIDATES; c++) { \
208 if ((T1)[c] >= NEKO_TUNE_INIT) { continue; } \
209 sprintf(neko_log_buf, "1D ch=%-4d: %9.2f us/call", \
210 NEKO_CHUNKS_SEL(LX, c), NEKO_TUNE_US((T1)[c], iters)); \
211 log_message(neko_log_buf); \
212 } \
213 } while (0)

215#define NEKO_TUNE_LOG_KSTEP(LX, T2) \
216 do { \
217 for (int c = 0; c < NEKO_EB_CANDIDATES; c++) { \

233/*
234 * TUNE-TIME VERIFICATION
235 *
236 * Every candidate the sweep is about to time is first run once on a synthetic
237 * input and compared with the reference formulation's result on the same
238 * input: the 1d kernel's first candidate where there is one, the kstep
239 * kernel's where there is not. A candidate whose max-norm difference exceeds
240 * NEKO_TUNE_VERIFY_TOL times the reference's max-norm is reported as REJECTED
241 * and excluded from the comparison, so it can be measured but cannot win.
242 *
243 * Why it exists: the matrix core contraction once ran for a long time with
244 * its lane layout wrong -- a wrong but symmetric operator makes CG converge
245 * happily to the solution of a different system (see mfma_contract_4x4 in
246 * mfma_kernel.h), so "the solver converges" never caught it. The single
247 * precision 4x4x1 tile added since carries a layout that is documented by
248 * AMD's calculator but has not been read out on hardware, and this is what
249 * keeps it honest: a wrong register index costs that candidate at tuning,
250 * loudly, not the solve.
251 *
252 * Why a synthetic input: the tuner runs on the first call of an operator,
253 * which can be on a zero field (a zero initial guess, a zero velocity), and
254 * every layout reproduces zero. The fields the operator differentiates are
255 * swapped for a deterministic non-zero pattern in (0.5, 1.5) for the duration
256 * of the pass -- the tune function hands over the addresses of its input
257 * pointer variables, which its launch macros name, so the swap is invisible
258 * to them -- while geometry, metrics and weights stay the case's own. The
259 * reference max-norm is logged with every rejection, so an inconclusive pass
260 * is visible as such.
261 *
262 * The tolerance is a layout detector, not an accuracy test: the variants sum
263 * the same terms in a different order, a few ulp per point, where a layout
264 * error reads as O(1). NaN or Inf anywhere in a candidate's output counts as
265 * an infinite difference, so it cannot slip through a comparison that is
266 * false for NaN.
267 *
268 * Cost: one launch per candidate plus a reduction over the output, once per
269 * (operator, order), and nin + nout arrays of n reals for the duration of the
270 * pass; a failed allocation skips the pass and says so in the log, it never
271 * stops the tune. NEKO_TUNE_VERIFY=0 turns it off, NEKO_TUNE_VERIFY_TOL
272 * overrides the tolerance.
273 */
274#ifndef NEKO_TUNE_VERIFY_DEFAULT
275#define NEKO_TUNE_VERIFY_DEFAULT 1
276#endif
277#define NEKO_TUNE_VERIFY_BLOCKS 256
278#define NEKO_TUNE_VERIFY_THREADS 256
279#define NEKO_TUNE_VERIFY_MAXIN 8
280/* Upper bound on any formulation's candidate count, for the reject flags */
281#define NEKO_TUNE_MAXCAND 16

303/* Deterministic non-zero pattern in (0.5, 1.5), different per input slot */
304static __global__ void neko_tune_fill_kernel(real * __restrict__ x,
305 const int n,
306 const unsigned int seed) {
307 for (int i = blockIdx.x * blockDim.x + threadIdx.x; i < n;
308 i += gridDim.x * blockDim.x) {
309 unsigned int h = (unsigned int) i * 2654435761u + seed * 40503u;
310 h ^= h >> 15;
311 h *= 2246822519u;
312 h ^= h >> 13;
313 x[i] = (real) 0.5 + (real) (h & 0xffffffu) / (real) 0x1000000u;
314 }
315}

317/* Per block maxima of |a - b| and |b|, NaN or Inf in a counted as infinite */
318static __global__ void neko_tune_diff_kernel(const real * __restrict__ a,
319 const real * __restrict__ b,
320 const int n,
321 double * __restrict__ part) {
322 __shared__ double sd[NEKO_TUNE_VERIFY_THREADS];
323 __shared__ double sr[NEKO_TUNE_VERIFY_THREADS];
324 double md = 0.0, mr = 0.0;
325
326 for (int i = blockIdx.x * blockDim.x + threadIdx.x; i < n;
327 i += gridDim.x * blockDim.x) {
328 const double x = (double) a[i];
329 const double y = (double) b[i];
330 const double d = fabs(x - y);
331 const double r = fabs(y);
332 if (!(d < HUGE_VAL)) {
333 md = HUGE_VAL;
334 } else if (d > md) {
335 md = d;
336 }
337 if (r > mr) {
338 mr = r;
339 }
340 }
341 sd[threadIdx.x] = md;
342 sr[threadIdx.x] = mr;
343 __syncthreads();
344 for (int s = blockDim.x / 2; s > 0; s >>= 1) {
345 if (threadIdx.x < s) {
346 if (sd[threadIdx.x + s] > sd[threadIdx.x]) {
347 sd[threadIdx.x] = sd[threadIdx.x + s];
348 }
349 if (sr[threadIdx.x + s] > sr[threadIdx.x]) {
350 sr[threadIdx.x] = sr[threadIdx.x + s];
351 }
352 }
353 __syncthreads();
354 }
355 if (threadIdx.x == 0) {
356 part[2 * blockIdx.x] = sd[0];
357 part[2 * blockIdx.x + 1] = sr[0];
358 }
359}

376/*
377 * Set the pass up: allocate, fill the synthetic inputs and point the tune
378 * function's input pointer variables at them. Returns 0, with nothing
379 * swapped, if the pass is off or cannot be set up.
380 */
381static int neko_tune_verify_begin(neko_tune_verify_t *v, real **outs,
382 const int nout, void ***ins, const int nin,
383 const int n, hipStream_t stream)
384{
385 const size_t bytes = (size_t) (nin + nout) * (size_t) n * sizeof(real);
386
387 if (!neko_tune_verify_on() || n <= 0 || nin > NEKO_TUNE_VERIFY_MAXIN) {
388 return 0;
389 }
390 v->outs = outs;
391 v->ins = ins;
392 v->nin = nin;
393 v->nout = nout;
394 v->n = n;
395 v->tol = neko_tune_verify_tol();
396 v->buf = NULL;
397 v->part = NULL;
398 if (hipMalloc((void **) &v->buf, bytes) != hipSuccess) {
399 /* Clear the sticky error: the tuner checks hipGetLastError() after
400 every launch and would otherwise report this one there */
401 (void) hipGetLastError();
402 return 0;
403 }
404 if (hipMalloc((void **) &v->part,
405 2 * NEKO_TUNE_VERIFY_BLOCKS * sizeof(double)) != hipSuccess) {
406 (void) hipGetLastError();
407 HIP_CHECK(hipFree(v->buf));
408 return 0;
409 }
410 for (int i = 0; i < nin; i++) {
411 real *x = v->buf + (size_t) i * n;
412 hipLaunchKernelGGL(neko_tune_fill_kernel, dim3(NEKO_TUNE_VERIFY_BLOCKS),
413 dim3(NEKO_TUNE_VERIFY_THREADS), 0, stream,
414 x, n, (unsigned int) (i + 1));
415 HIP_CHECK(hipGetLastError());
416 v->saved[i] = *ins[i];
417 *ins[i] = (void *) x;
418 }
419 return 1;
420}

422/* Snapshot the outputs of the reference launch the caller has just made */
423static void neko_tune_verify_ref(neko_tune_verify_t *v, hipStream_t stream)
424{
425 for (int j = 0; j < v->nout; j++) {
426 /* hipMemcpyDefault rather than DeviceToDevice: the runtime then
427 resolves the kind from the pointers, which keeps this correct if
428 the operator's arrays are ever host-mapped on an APU */
429 HIP_CHECK(hipMemcpyAsync(v->buf + (size_t) (v->nin + j) * v->n,
430 v->outs[j], (size_t) v->n * sizeof(real),
431 hipMemcpyDefault, stream));
432 }
433 HIP_CHECK(hipStreamSynchronize(stream));
434}

436/*
437 * Compare the outputs of the launch the caller has just made with the
438 * snapshot. Returns 1 if every output is within tolerance, and leaves the
439 * worst max-norm difference and the reference max-norm behind for the log
440 * either way. Written so that an infinite difference fails.
441 */
442static int neko_tune_verify_check(neko_tune_verify_t *v, hipStream_t stream,
443 double *maxdiff, double *maxref)
444{
445 double part[2 * NEKO_TUNE_VERIFY_BLOCKS];
446
447 *maxdiff = 0.0;
448 *maxref = 0.0;
449 for (int j = 0; j < v->nout; j++) {
450 const real *ref = v->buf + (size_t) (v->nin + j) * v->n;
451 hipLaunchKernelGGL(neko_tune_diff_kernel, dim3(NEKO_TUNE_VERIFY_BLOCKS),
452 dim3(NEKO_TUNE_VERIFY_THREADS), 0, stream,
453 (const real *) v->outs[j], ref, v->n, v->part);
454 HIP_CHECK(hipGetLastError());
455 HIP_CHECK(hipMemcpyAsync(part, v->part, sizeof(part),
456 hipMemcpyDeviceToHost, stream));
457 HIP_CHECK(hipStreamSynchronize(stream));
458 for (int b = 0; b < NEKO_TUNE_VERIFY_BLOCKS; b++) {
459 if (part[2 * b] > *maxdiff) {
460 *maxdiff = part[2 * b];
461 }
462 if (part[2 * b + 1] > *maxref) {
463 *maxref = part[2 * b + 1];
464 }
465 }
466 }
467 return (*maxdiff <= v->tol * (*maxref));
468}
