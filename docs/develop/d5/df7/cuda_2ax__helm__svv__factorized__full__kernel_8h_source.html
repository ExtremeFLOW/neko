61template< typename T >
62__device__ __forceinline__ void
63ax_helm_svv_factorized_stress_flux(const T ur, const T us, const T ut,
64 const T vr, const T vs, const T vt,
65 const T wr, const T ws, const T wt,
66 const T drdx, const T drdy, const T drdz,
67 const T dsdx, const T dsdy, const T dsdz,
68 const T dtdx, const T dtdy, const T dtdz,
69 const T dj,
70 T & shur, T & shus,
71 T & shvr, T & shvs,
72 T & shwr, T & shws,
73 T & rut, T & rvt, T & rwt) {
74
75 const T u1 = ur * drdx + us * dsdx + ut * dtdx;
76 const T u2 = ur * drdy + us * dsdy + ut * dtdy;
77 const T u3 = ur * drdz + us * dsdz + ut * dtdz;
78
79 const T v1 = vr * drdx + vs * dsdx + vt * dtdx;
80 const T v2 = vr * drdy + vs * dsdy + vt * dtdy;
81 const T v3 = vr * drdz + vs * dsdz + vt * dtdz;
82
83 const T w1 = wr * drdx + ws * dsdx + wt * dtdx;
84 const T w2 = wr * drdy + ws * dsdy + wt * dtdy;
85 const T w3 = wr * drdz + ws * dsdz + wt * dtdz;
86
87 const T s11 = dj * (u1 + u1);
88 const T s22 = dj * (v2 + v2);
89 const T s33 = dj * (w3 + w3);
90 const T s12 = dj * (u2 + v1);
91 const T s13 = dj * (u3 + w1);
92 const T s23 = dj * (v3 + w2);
93
94 shur = drdx * s11 + drdy * s12 + drdz * s13;
95 shus = dsdx * s11 + dsdy * s12 + dsdz * s13;
96 rut = dtdx * s11 + dtdy * s12 + dtdz * s13;
97
98 shvr = drdx * s12 + drdy * s22 + drdz * s23;
99 shvs = dsdx * s12 + dsdy * s22 + dsdz * s23;
100 rvt = dtdx * s12 + dtdy * s22 + dtdz * s23;
101
102 shwr = drdx * s13 + drdy * s23 + drdz * s33;
103 shws = dsdx * s13 + dsdy * s23 + dsdz * s33;
104 rwt = dtdx * s13 + dtdy * s23 + dtdz * s33;
105}
