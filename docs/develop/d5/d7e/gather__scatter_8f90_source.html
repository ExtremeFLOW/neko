176 subroutine gs_init(gs, dofmap, bcknd, comm_bcknd)
177 class(gs_t), intent(inout) :: gs
178 type(dofmap_t), target, intent(inout) :: dofmap
179 character(len=LOG_SIZE) :: log_buf
180 character(len=20) :: bcknd_str
181 integer, optional :: bcknd, comm_bcknd
182 integer :: i, ierr, bcknd_, comm_bcknd_
183 integer(i8) :: glb_nshared, glb_nlocal
184 logical :: use_device_mpi, use_device_nccl, use_device_shmem, use_host_mpi
185 logical :: use_host_shmem
186 logical :: use_caf
187 logical :: use_neighbour
188 logical :: use_utofu
189 logical :: use_mpi_rma
190 logical :: tune_comm
191 real(kind=rp), allocatable :: tmp(:)
192 type(c_ptr) :: tmp_d = c_null_ptr
193 integer :: strtgy(4) = [int(b'00'), int(b'01'), int(b'10'), int(b'11')]
194 integer :: avg_strtgy, env_len
195 character(len=255) :: env_strtgy, env_gscomm
196 real(kind=dp) :: strtgy_time(4)
197
198 call gs%free()
199
200 call neko_log%section('Gather-Scatter')
201 ! Currently this uses the dofmap which also contains geometric information
202 ! Only connectivity/numbering of points is technically necessary for gs
203 gs%dofmap => dofmap
204
205 use_device_mpi = .false.
206 use_device_nccl = .false.
207 use_device_shmem = .false.
208 use_host_mpi = .false.
209 use_host_shmem = .false.
210 use_caf = .false.
211 use_neighbour = .false.
212 use_utofu = .false.
213 use_mpi_rma = .false.
214 tune_comm = .false.
215
216 ! Check if a comm-backend is requested via env. variables
217 call get_environment_variable("NEKO_GS_COMM", env_gscomm, env_len)
218 if (env_len .gt. 0) then
219 if (env_gscomm(1:env_len) .eq. "MPI") then
220 use_host_mpi = .true.
221 else if (env_gscomm(1:env_len) .eq. "MPIGPU") then
222 use_device_mpi = .true.
223 else if (env_gscomm(1:env_len) .eq. "NCCL") then
224 use_device_nccl = .true.
225 else if (env_gscomm(1:env_len) .eq. "SHMEM") then
226 if (neko_bcknd_device .eq. 1) then
227 use_device_shmem = .true.
228 else
229 use_host_shmem = .true.
230 end if
231 else if (env_gscomm(1:env_len) .eq. "CAF") then
232 use_caf = .true.
233 else if (env_gscomm(1:env_len) .eq. "NEIGHBOUR" .or. &
234 env_gscomm(1:env_len) .eq. "NEIGHBOR") then
235 use_neighbour = .true.
236 else if (env_gscomm(1:env_len) .eq. "UTOFU") then
237 use_utofu = .true.
238 else if (env_gscomm(1:env_len) .eq. "MPIRMA" .or. &
239 env_gscomm(1:env_len) .eq. "RMA") then
240 use_mpi_rma = .true.
241 else
242 call neko_error('Unknown Gather-scatter comm. backend')
243 end if
244 end if
245
246
247 if (present(comm_bcknd)) then
248 comm_bcknd_ = comm_bcknd
249 else if (use_host_mpi) then
250 comm_bcknd_ = gs_comm_mpi
251 else if (use_device_mpi) then
252 comm_bcknd_ = gs_comm_mpigpu
253 else if (use_device_nccl) then
254 comm_bcknd_ = gs_comm_nccl
255 else if (use_device_shmem) then
256 comm_bcknd_ = gs_comm_nvshmem
257 else if (use_host_shmem) then
258 comm_bcknd_ = gs_comm_openshmem
259 else if (use_caf) then
260 comm_bcknd_ = gs_comm_caf
261 else if (use_neighbour) then
262 comm_bcknd_ = gs_comm_neighbour
263 else if (use_utofu) then
264 comm_bcknd_ = gs_comm_utofu
265 else if (use_mpi_rma) then
266 comm_bcknd_ = gs_comm_mpirma
267 else
268 if (neko_device_mpi) then
269 comm_bcknd_ = gs_comm_mpigpu
270 use_device_mpi = .true.
271 else
272 ! No backend requested, benchmark the host backends once the
273 ! schedule is known and keep the fastest one (see gs_tune_comm)
274 comm_bcknd_ = gs_comm_mpi
275 tune_comm = (pe_size .gt. 1)
276 end if
277 end if
278
279 call gs_comm_alloc(gs%comm, comm_bcknd_)
280
281 if (tune_comm) then
282 call neko_log%message('Comm : auto')
283 else
284 call neko_log%message('Comm : ' // gs_comm_name(comm_bcknd_))
285 end if
286 ! Initialize a stack for each rank containing which dofs to send/recv at
287 ! that rank
288 call gs%comm%init_dofs()
289 ! Initialize mapping between local ids and gather-scatter ids
290 ! based on the global numbering in dofmap
291 call gs_init_mapping(gs)
292 ! Setup buffers and which ranks to send/recv data from based on mapping
293 ! and initializes gs%comm (sets up gs%comm%send_dof and gs%comm%recv_dof and
294 ! recv_pe/send_pe)
295 call gs_schedule(gs)
296 ! Global number of points not needing to be sent over mpi for gs operations
297 ! "Internal points"
298 glb_nlocal = int(gs%nlocal, i8)
299 ! Global number of points needing to be communicated with other pes/ranks
300 ! "external points"
301 glb_nshared = int(gs%nshared, i8)
302 ! Can be thought of a measure of the volume of this rank (glb_nlocal) and
303 ! the surface area (glb_nshared) that is shared with other ranks
304 ! Lots of internal volume compared to surface that needs communication is
305 ! good
306
307 if (pe_rank .eq. 0) then
308 call mpi_reduce(mpi_in_place, glb_nlocal, 1, &
309 mpi_integer8, mpi_sum, 0, neko_comm, ierr)
310
311 call mpi_reduce(mpi_in_place, glb_nshared, 1, &
312 mpi_integer8, mpi_sum, 0, neko_comm, ierr)
313 else
314 call mpi_reduce(glb_nlocal, glb_nlocal, 1, &
315 mpi_integer8, mpi_sum, 0, neko_comm, ierr)
316
317 call mpi_reduce(glb_nshared, glb_nshared, 1, &
318 mpi_integer8, mpi_sum, 0, neko_comm, ierr)
319 end if
320
321 write(log_buf, '(A,I12)') 'Avg. internal: ', glb_nlocal/pe_size
322 call neko_log%message(log_buf)
323 write(log_buf, '(A,I12)') 'Avg. external: ', glb_nshared/pe_size
324 call neko_log%message(log_buf)
325
326 if (present(bcknd)) then
327 bcknd_ = bcknd
328 else
329 if (neko_bcknd_sx .eq. 1) then
330 bcknd_ = gs_bcknd_sx
331 else if (neko_bcknd_device .eq. 1) then
332 bcknd_ = gs_bcknd_dev
333 else
334 bcknd_ = gs_bcknd_cpu
335 end if
336 end if
337
338 ! Setup Gather-scatter backend
339 select case (bcknd_)
340 case (gs_bcknd_cpu)
341 allocate(gs_cpu_t::gs%bcknd)
342 bcknd_str = ' std'
343 case (gs_bcknd_dev)
344 allocate(gs_device_t::gs%bcknd)
345 if (neko_bcknd_hip .eq. 1) then
346 bcknd_str = ' hip'
347 else if (neko_bcknd_cuda .eq. 1) then
348 bcknd_str = ' cuda'
349 else if (neko_bcknd_opencl .eq. 1) then
350 bcknd_str = ' opencl'
351 else if (neko_bcknd_metal .eq. 1) then
352 bcknd_str = ' metal'
353 end if
354 case (gs_bcknd_sx)
355 allocate(gs_sx_t::gs%bcknd)
356 bcknd_str = ' sx'
357 case default
358 call neko_error('Unknown Gather-scatter backend')
359 end select
360
361 write(log_buf, '(A)') 'Backend : ' // trim(bcknd_str)
362 call neko_log%message(log_buf)
363
364
365 call gs%bcknd%init(gs%nlocal, gs%nshared, gs%nlocal_blks, gs%nshared_blks)
366
367 ! Plain base-type assignment; setting this through a
368 ! select type (gs_device_t) miscompiles with CCE 21 at -O2/-O3,
369 ! silently leaving shared points on the host so that the scatter
370 ! overwrites the unpacked halo data with the stale host buffer
371 if (use_device_mpi .or. use_device_nccl .or. use_device_shmem) then
372 gs%bcknd%shared_on_host = .false.
373 end if
374
375 if (use_device_mpi) then
376 if (pe_size .gt. 1) then
377 ! Select fastest device MPI strategy at runtime
378 select type (c => gs%comm)
379 type is (gs_device_mpi_t)
380 call get_environment_variable("NEKO_GS_STRTGY", env_strtgy, &
381 env_len)
382 if (env_len .eq. 0) then
383 allocate(tmp(dofmap%size()))
384 call device_map(tmp, tmp_d, dofmap%size())
385 tmp = 1.0_rp
386 call device_memcpy(tmp, tmp_d, dofmap%size(), &
387 host_to_device, sync = .false.)
388
389 do i = 1, size(strtgy)
390 c%nb_strtgy = strtgy(i)
391 strtgy_time(i) = gs_time_ops(gs, tmp, dofmap%size(), &
392 gs_op_add, gs_tune_ntrials)
393 end do
394
395 call device_unmap(tmp, tmp_d)
396 deallocate(tmp)
397
398 c%nb_strtgy = strtgy(minloc(strtgy_time, 1))
399
400 avg_strtgy = minloc(strtgy_time, 1)
401 call mpi_allreduce(mpi_in_place, avg_strtgy, 1, &
402 mpi_integer, mpi_sum, neko_comm)
403 avg_strtgy = avg_strtgy / pe_size
404
405 write(log_buf, '(A,B0.2,A)') 'Avg. strtgy : [', &
406 strtgy(avg_strtgy), ']'
407
408 else
409 read(env_strtgy(1:env_len), *) i
410
411 if (i .lt. 1 .or. i .gt. 4) then
412 call neko_error('Invalid gs sync strtgy')
413 end if
414
415 c%nb_strtgy = strtgy(i)
416 avg_strtgy = i
417
418 write(log_buf, '(A,B0.2,A)') 'Env. strtgy : [', &
419 strtgy(avg_strtgy), ']'
420 end if
421
422 call neko_log%message(log_buf)
423
424 end select
425 end if
426 end if
427
428 ! Select the fastest host comm. backend at runtime
429 if (tune_comm) then
430 call gs_tune_comm(gs, dofmap%size(), comm_bcknd_)
431 end if
432
433 call neko_log%end_section()
434
435 end subroutine gs_init

508 subroutine gs_free(gs)
509 class(gs_t), intent(inout) :: gs
510
511 nullify(gs%dofmap)
512
513 ! The device backend lazily maps these arrays in gather/scatter,
514 ! and its free only releases the device side; remove any stale
515 ! address table entries before deallocating the host side
516 if (allocated(gs%local_gs)) then
517 if (neko_bcknd_device .eq. 1) then
518 call device_deassociate(gs%local_gs)
519 end if
520 deallocate(gs%local_gs)
521 end if
522
523 if (allocated(gs%local_dof_gs)) then
524 if (neko_bcknd_device .eq. 1) then
525 call device_deassociate(gs%local_dof_gs)
526 end if
527 deallocate(gs%local_dof_gs)
528 end if
529
530 if (allocated(gs%local_gs_dof)) then
531 if (neko_bcknd_device .eq. 1) then
532 call device_deassociate(gs%local_gs_dof)
533 end if
534 deallocate(gs%local_gs_dof)
535 end if
536
537 if (allocated(gs%local_blk_len)) then
538 if (neko_bcknd_device .eq. 1) then
539 call device_deassociate(gs%local_blk_len)
540 end if
541 deallocate(gs%local_blk_len)
542 end if
543
544 if (allocated(gs%local_blk_off)) then
545 if (neko_bcknd_device .eq. 1) then
546 call device_deassociate(gs%local_blk_off)
547 end if
548 deallocate(gs%local_blk_off)
549 end if
550
551 if (allocated(gs%shared_gs)) then
552 if (neko_bcknd_device .eq. 1) then
553 call device_deassociate(gs%shared_gs)
554 end if
555 deallocate(gs%shared_gs)
556 end if
557
558 if (allocated(gs%shared_gs_v)) then
559 if (neko_bcknd_device .eq. 1 .and. c_associated(gs%shared_gs_v_d)) then
560 call device_unmap(gs%shared_gs_v, gs%shared_gs_v_d)
561 end if
562 deallocate(gs%shared_gs_v)
563 end if
564
565 if (allocated(gs%shared_dof_gs)) then
566 if (neko_bcknd_device .eq. 1) then
567 call device_deassociate(gs%shared_dof_gs)
568 end if
569 deallocate(gs%shared_dof_gs)
570 end if
571
572 if (allocated(gs%shared_gs_dof)) then
573 if (neko_bcknd_device .eq. 1) then
574 call device_deassociate(gs%shared_gs_dof)
575 end if
576 deallocate(gs%shared_gs_dof)
577 end if
578
579 if (allocated(gs%shared_blk_len)) then
580 if (neko_bcknd_device .eq. 1) then
581 call device_deassociate(gs%shared_blk_len)
582 end if
583 deallocate(gs%shared_blk_len)
584 end if
585
586 if (allocated(gs%shared_blk_off)) then
587 if (neko_bcknd_device .eq. 1) then
588 call device_deassociate(gs%shared_blk_off)
589 end if
590 deallocate(gs%shared_blk_off)
591 end if
592
593 gs%nlocal = 0
594 gs%nshared = 0
595 gs%nlocal_blks = 0
596 gs%nshared_blks = 0
597
598 call gs%shared_dofs%free()
599
600 if (allocated(gs%bcknd)) then
601 call gs%bcknd%free()
602 deallocate(gs%bcknd)
603 end if
604
605 if (allocated(gs%comm)) then
606 call gs%comm%free()
607 deallocate(gs%comm)
608 end if
609
610 end subroutine gs_free

613 subroutine gs_init_mapping(gs)
614 type(gs_t), target, intent(inout) :: gs
615 type(mesh_t), pointer :: msh
616 type(dofmap_t), pointer :: dofmap
617 type(stack_i4_t), target :: local_dof, dof_local, shared_dof, dof_shared

1488 subroutine gs_schedule(gs)
1489 type(gs_t), target, intent(inout) :: gs
1490 type(htable_iter_i8_t) :: it
1491 type(stack_i4_t) :: send_pe, recv_pe
1492 type(stack_i8_t) :: cr_buf
1493 integer(i8), allocatable :: buf(:)
1494 integer(i8), pointer :: cr_data(:)
1495 integer(i8), allocatable :: rgid(:), gtmp(:)
1496 integer, allocatable :: rpeer(:), rgsid(:), rperm(:), gperm(:)
1497 integer(i8) :: gid
1498 integer :: i, j, n, owner, nrec, peer, shared_gs_id, tmp
1499 integer :: a, b, cnt, t
1500
1501 call send_pe%init()
1502 call recv_pe%init()
1503
1504 !
1505 ! Phase 1: route every local shared dof to its canonical owner.
1506 ! record = [dest=owner, len=2, gid, origin]
1507 !
1508 call cr_buf%init(max(gs%shared_dofs%num_entries(), 1) * 4)
1509 call it%init(gs%shared_dofs)
1510 do while (it%next())
1511 gid = it%key()
1512 owner = int(modulo(gid, int(pe_size, i8)))
1513 call crystal_router_pack(cr_buf, owner, [gid, int(pe_rank, i8)])
1514 end do
1515
1516 n = cr_buf%size()
1517 allocate(buf(max(n, 1)))
1518 if (n .gt. 0) then
1519 cr_data => cr_buf%array()
1520 buf(1:n) = cr_data(1:n)
1521 end if
1522 call cr_buf%free()
1523
1524 call crystal_router_transfer(buf, n)
1525
1526 !
1527 ! Phase 2: at the owner, group holders by gid and reflect, to each holder,
1528 ! every other holder of the same dof.
1529 ! reply = [dest=holder, len=2, gid, peer]
1530 !
1531 nrec = n / 4 ! every record here has the fixed form [me, 2, gid, origin]
1532 allocate(rgid(max(nrec, 1)), rgsid(max(nrec, 1)), gperm(max(nrec, 1)))
1533 do i = 1, nrec
1534 rgid(i) = buf((i - 1) * 4 + 3) ! gid
1535 rgsid(i) = int(buf((i - 1) * 4 + 4)) ! origin rank (reuse array)
1536 end do
1537 if (nrec .gt. 0) call gs_sort_i8(rgid, gperm, nrec)
1538
1539 call cr_buf%init(max(n, 1))
1540 i = 1
1541 do while (i .le. nrec)
1542 j = i
1543 do while (j .le. nrec)
1544 if (rgid(j) .ne. rgid(i)) exit
1545 j = j + 1
1546 end do
1547 ! Reflect, to each holder, every other holder of this dof.
1548 if (j - i .gt. 1) then
1549 do a = i, j - 1 ! recipient holder
1550 do b = i, j - 1 ! the other holder
1551 if (a .eq. b) cycle
1552 call crystal_router_pack(cr_buf, rgsid(gperm(a)), &
1553 [rgid(i), int(rgsid(gperm(b)), i8)])
1554 end do
1555 end do
1556 end if
1557 i = j
1558 end do
1559 deallocate(rgid, rgsid, gperm)
1560
1561 n = cr_buf%size()
1562 if (allocated(buf)) deallocate(buf)
1563 allocate(buf(max(n, 1)))
1564 if (n .gt. 0) then
1565 cr_data => cr_buf%array()
1566 buf(1:n) = cr_data(1:n)
1567 end if
1568 call cr_buf%free()
1569
1570 call crystal_router_transfer(buf, n)
1571
1572 !
1573 ! Phase 3: register each (dof, peer) for both send and receive. Order each
1574 ! peer's dof list by gid so both ranks of a pair agree on the order.
1575 !
1576 nrec = n / 4 ! replies are [me, 2, gid, peer]
1577 allocate(rgid(max(nrec, 1)), rpeer(max(nrec, 1)), rgsid(max(nrec, 1)), &
1578 rperm(max(nrec, 1)))
1579 do i = 1, nrec
1580 gid = buf((i - 1) * 4 + 3)
1581 rgid(i) = gid
1582 rpeer(i) = int(buf((i - 1) * 4 + 4))
1583 tmp = gs%shared_dofs%get(gid, shared_gs_id)
1584 rgsid(i) = shared_gs_id
1585 end do
1586
1587 ! Sort by peer; within each peer run, sort by gid and register in that order.
1588 if (nrec .gt. 0) call sort(rpeer, rperm, nrec)
1589 a = 1
1590 do while (a .le. nrec)
1591 b = a
1592 do while (b .le. nrec)
1593 if (rpeer(b) .ne. rpeer(a)) exit
1594 b = b + 1
1595 end do
1596 peer = rpeer(a)
1597 cnt = b - a
1598 allocate(gtmp(cnt), gperm(cnt))
1599 do t = 1, cnt
1600 gtmp(t) = rgid(rperm(a + t - 1))
1601 end do
1602 call gs_sort_i8(gtmp, gperm, cnt)
1603 do t = 1, cnt
1604 shared_gs_id = rgsid(rperm(a + gperm(t) - 1))
1605 call gs%comm%send_dof(peer)%push(shared_gs_id)
1606 call gs%comm%recv_dof(peer)%push(shared_gs_id)
1607 end do
1608 deallocate(gtmp, gperm)
1609 call send_pe%push(peer)
1610 call recv_pe%push(peer)
1611 a = b
1612 end do
1613 deallocate(rgid, rpeer, rgsid, rperm)
1614 if (allocated(buf)) deallocate(buf)
1615
1616 call gs%comm%init(send_pe, recv_pe)
1617
1618 call send_pe%free()
1619 call recv_pe%free()
1620
1621 !This arrays seems to take massive amounts of memory...
1622 call gs%shared_dofs%free()
1623
1624 end subroutine gs_schedule
