56static inline id<MTLComputePipelineState>
57neko_metal_pipeline(NSString *name) {
58 if (!neko_metal_pipelines) {
59 neko_metal_pipelines = [NSMutableDictionary new];
60 }
61 id<MTLComputePipelineState> pso = neko_metal_pipelines[name];
62 if (!pso) {
63 id<MTLDevice> dev = neko_metal_device();
64 NSError *err = nil;
65 id<MTLLibrary> lib = neko_metal_library();
66 id<MTLFunction> fn = [lib newFunctionWithName:name];
67 if (!fn) {
68 NSLog(@"Metal: kernel '%@' not found in library", name);
69 abort();
70 }
71 pso = [dev newComputePipelineStateWithFunction:fn error:&err];
72 if (err) {
73 NSLog(@"Metal: pipeline error for '%@': %@", name, err);
74 abort();
75 }
76 neko_metal_pipelines[name] = pso;
77 }
78 return pso;
79}

91static inline void
92neko_metal_dispatch_1d_queue(id<MTLCommandQueue> queue,
93 id<MTLComputePipelineState> pso,
94 void (^encode)(id<MTLComputeCommandEncoder>),
95 NSUInteger n) {
96 if (n == 0) return;
97 @autoreleasepool {
98 id<MTLCommandBuffer> cb = [queue commandBuffer];
99 id<MTLComputeCommandEncoder> enc = [cb computeCommandEncoder];
100 [enc setComputePipelineState:pso];
101 encode(enc);
102 MTLSize threads = MTLSizeMake(n, 1, 1);
103 NSUInteger tpg = n < NEKO_METAL_NTHREADS ? n : NEKO_METAL_NTHREADS;
104 MTLSize tg = MTLSizeMake(tpg, 1, 1);
105 [enc dispatchThreads:threads threadsPerThreadgroup:tg];
106 [enc endEncoding];
107 [cb commit];
108 [cb waitUntilCompleted];
109 }
110}

117static inline void
118neko_metal_dispatch_groups(id<MTLComputePipelineState> pso,
119 void (^encode)(id<MTLComputeCommandEncoder>),
120 NSUInteger ngroups, NSUInteger tpg) {
121 if (ngroups == 0) return;
122 @autoreleasepool {
123 id<MTLCommandQueue> queue =
124 (__bridge id<MTLCommandQueue>) glb_cmd_queue;
125 id<MTLCommandBuffer> cb = [queue commandBuffer];
126 id<MTLComputeCommandEncoder> enc = [cb computeCommandEncoder];
127 [enc setComputePipelineState:pso];
128 encode(enc);
129 [enc dispatchThreadgroups:MTLSizeMake(ngroups, 1, 1)
130 threadsPerThreadgroup:MTLSizeMake(tpg, 1, 1)];
131 [enc endEncoding];
132 [cb commit];
133 [cb waitUntilCompleted];
134 }
135}
