100 subroutine user_stats_init_from_json(this, json, case)
101 class(user_stats_t), intent(inout), target :: this
102 type(json_file), intent(inout) :: json
103 class(case_t), intent(inout), target :: case
104 character(len=:), allocatable :: filename
105 character(len=:), allocatable :: avg_dir
106 character(len=:), allocatable :: name
107
108 call json_get_or_default(json, "name", name, "user_stats")
109 call this%init_base(json, case)
110
112 call json%info('fields', n_children = this%n_avg_fields)
113 call json_get(json, 'fields', this%field_names)
114 call json_get_or_lookup_or_default(json, 'start_time', this%start_time, &
115 0.0_dp)
116 call json_get_or_default(json, 'avg_direction', avg_dir, 'none')
117
118 if (json%valid_path('output_filename')) then
119 call json_get(json, 'output_filename', filename)
120 call user_stats_init_common(this, name, this%start_time, &
121 case%fluid%c_Xh, avg_dir, filename = filename)
122 else
123 call user_stats_init_common(this, name, this%start_time, &
124 case%fluid%c_Xh, avg_dir)
125 end if
126
127 end subroutine user_stats_init_from_json

130 class(user_stats_t), intent(inout) :: this
131 type(time_state_t), intent(in) :: time
132 character(len=NEKO_FNAME_LEN) :: fname

235 subroutine user_stats_init_common(this, name, start_time, coef, avg_dir, &
236 filename, precision)
237 class(user_stats_t), intent(inout) :: this
238 character(len=*), intent(in) :: name
239 character(len=*), intent(in), optional :: filename
240 integer, intent(in), optional :: precision
241 real(kind=dp), intent(in) :: start_time
242 character(len=*), intent(in) :: avg_dir
243 type(coef_t), intent(inout) :: coef
244 integer :: i
245 type(field_t), pointer :: field_to_avg
246 character(len=NEKO_FNAME_LEN) :: stats_fname
247 character(len=1024) :: unique_name
248
249 unique_name = name // "/"
250
251 this%name = name
252 this%start_time = start_time
253 this%time = start_time
254
255 if (present(filename)) then
256 this%base_filename = filename
257 else
258 this%base_filename = "user_stats"
259 end if
260 stats_fname = trim(this%base_filename) // "0"
261
263 allocate(this%mean_fields(this%n_avg_fields))
264 do i = 1, this%n_avg_fields
265 field_to_avg => neko_registry%get_field(trim(this%field_names(i)))
266 call this%mean_fields(i)%init(field_to_avg, trim(unique_name) // &
267 "mean_" // trim(this%field_names(i)))
268 end do
269
270 call this%output%init(this%mean_fields, this%n_avg_fields, &
271 this%start_time, coef, avg_dir, name = stats_fname, &
272 path = this%case%output_directory)
273 call this%case%output_controller%add(this%output, &
274 this%output_controller%control_value, &
275 this%output_controller%control_mode)
276
277 nullify(field_to_avg)
278
279 end subroutine user_stats_init_common

303 subroutine user_stats_compute(this, time)
304 class(user_stats_t), intent(inout) :: this
305 type(time_state_t), intent(in) :: time
306 integer :: i
307
309 if (time%t .ge. this%start_time) then
310 do i = 1, this%n_avg_fields
311 call this%mean_fields(i)%update( real(time%t - this%time, kind=rp) )
312 end do
313 this%time = time%t
314 end if

316 end subroutine user_stats_compute
