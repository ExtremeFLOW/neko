115/*
116 * Was sm_90 among the architectures this translation unit was compiled for?
117 * Same trap as dmma_arch_compiled() in dmma_kernel.h: a binary built for an
118 * older arch still runs on Hopper by JIT compiling its PTX, and the kernel
119 * body is guarded on __CUDA_ARCH__, so without this check a build for, say,
120 * sm_80 running on a GH200 would JIT the *no-op* body, time as free, win the
121 * tuner comparison and leave a stale w. Unlike the DMMA check this one is
122 * sm_90 exactly, not a range: sm_100 and later move the bulk copy semantics
123 * and would have to be measured before being allowed in.
124 */
125#ifndef NEKO_TMA_ARCH_COMPILED
126#if defined(__CUDA_ARCH_LIST__)
127static inline bool tma_arch_compiled()
128{
129 const int arch[] = { __CUDA_ARCH_LIST__ };
130
131 for (int i = 0; i < (int) (sizeof(arch)/sizeof(arch[0])); i++) {
132 if (arch[i] >= 900 && arch[i] < 1000) {
133 return true;
134 }
135 }
136 return false;
137}
138#else
139static inline bool tma_arch_compiled()
140{
141 return false;
142}
143#endif
144#else
145static inline bool tma_arch_compiled()
146{
147 return (NEKO_TMA_ARCH_COMPILED != 0);
148}

209/*
210 * The batched vector variant's block, as one struct in dynamic shared memory.
211 *
212 * Why a struct and why dynamic. The measurement that motivates this variant is
213 * that a bulk copy only pays for itself in company: the scalar kernel issues
214 * eight at once and wins at 20 warps per SM against 48, while the
215 * component-at-a-time vector kernel issues seven once and then two lone 4 kB
216 * copies, and loses at 20 warps against 16 -- with MORE occupancy than the
217 * kernel it loses to, which is what rules residency out as the cause. Putting
218 * every one of an element's thirteen cubes in shared at once takes the block
219 * to 54800 B, past the 48 kB a block gets without opting in, so the allocation
220 * has to be dynamic and the layout has to live somewhere. A struct puts it in
221 * one place and makes the launch size sizeof(), which cannot drift from it.
222 *
223 * Member order is the layout: every cube lands on a 4 kB boundary and the two
224 * mbarriers go last, so no padding is inserted and every bulk copy target is
225 * aligned far past the 16 bytes it needs. The asserts below hold that.
226 */
227struct dmma_tma_batch_smem {
228 double c[3][DMMA_CUBE]; /* the three components, in and then out */
229 double r[DMMA_CUBE]; /* reference derivatives, reused by each */
230 double s[DMMA_CUBE]; /* component in turn */
231 double t[DMMA_CUBE];
232 double g[DMMA_NG][DMMA_CUBE]; /* h1, g11, g22, g33, g12, g13, g23 */
233 double dx[DMMA_MAT];
234 double dy[DMMA_MAT];
235 double dz[DMMA_MAT];
236 unsigned long long bar_a; /* the first component's cube */
237 unsigned long long bar_b; /* the other two, and the seven factors */
238};
239
240#define NEKO_DMMA_TMA_BATCH_SMEM ((int) sizeof(dmma_tma_batch_smem))
241
242/* 227 kB is the most a Hopper block can opt into; the runtime gate below
243 checks the device rather than trusting this */
244static_assert(sizeof(dmma_tma_batch_smem) <= 227 * 1024,
245 "dmma tma batch block exceeds the opt-in shared memory ceiling");
246static_assert(offsetof(dmma_tma_batch_smem, c) % 128 == 0 &&
247 offsetof(dmma_tma_batch_smem, g) % 128 == 0,
248 "dmma tma batch bulk copy targets must stay 128 byte aligned");
249
256static inline bool cuda_have_tma_batch()
257{
258#if NEKO_TMA_TOOLKIT
259 static int cached = -1;
260
261 if (cached < 0) {
262 int dev = 0;
263 int optin = 0;
264
265 if (!cuda_have_tma()) {
266 cached = 0;
267 } else if (cudaGetDevice(&dev) == cudaSuccess &&
268 cudaDeviceGetAttribute(&optin,
269 cudaDevAttrMaxSharedMemoryPerBlockOptin,
270 dev) == cudaSuccess) {
271 cached = (optin >= NEKO_DMMA_TMA_BATCH_SMEM) ? 1 : 0;
272 } else {
273 cached = 0;
274 }
275 }
276 return cached == 1;
277#else
278 return false;
279#endif
280}

379/*
380 * The mbarrier and bulk copy primitives, as inline PTX.
381 *
382 * Written out rather than reached through cuda::memcpy_async() and
383 * cuda::barrier because the point of the variant is to measure the TMA
384 * engine: libcu++ selects cp.async.bulk only when it can prove the size and
385 * alignment, and falls back to cp.async otherwise, which would still compile,
386 * still be correct, and quietly measure something else entirely.
387 *
388 * Every mbarrier and shared operand is addressed in the shared window, which
389 * is what __cvta_generic_to_shared() produces.
390 */
391
392__device__ __forceinline__ static unsigned int tma_smem(const void *p)
393{
394 return (unsigned int) __cvta_generic_to_shared(p);
395}
396
397/* And the global window, likewise. The bulk copies name the global state
398 space explicitly, so a generic address is not theirs to reinterpret: the
399 conversion is written out rather than left to the pointer happening to be
400 global already */
401__device__ __forceinline__ static unsigned long long tma_gmem(const void *p)
402{
403 return (unsigned long long) __cvta_generic_to_global(p);
404}
405
406/* 'count' is the number of arrivals the barrier waits for, one per producer
407 thread. Must be visible to every waiter before any of them waits, i.e. a
408 __syncthreads() has to separate this from the first tma_wait() */
409__device__ __forceinline__ static void tma_barrier_init(unsigned long long *bar,
410 const int count)
411{
412 asm volatile("mbarrier.init.shared::cta.b64 [%0], %1;"
413 :: "r"(tma_smem(bar)), "r"(count) : "memory");
414}
415
416/* Arrive, and declare how many bytes the copies about to be issued will
417 deliver. The barrier flips phase once the arrival has landed *and* the
418 completing copies have paid off the transaction count, so this has to
419 precede the copies it accounts for */
420__device__ __forceinline__ static void tma_expect(unsigned long long *bar,
421 const int bytes)
422{
423 asm volatile("mbarrier.arrive.expect_tx.shared::cta.b64 _, [%0], %1;"
424 :: "r"(tma_smem(bar)), "r"(bytes) : "memory");
425}
426
427/* One contiguous global -> shared bulk copy, completing on 'bar' */
428__device__ __forceinline__ static void tma_load(void *dst, const void *src,
429 const int bytes,
430 unsigned long long *bar)
431{
432 asm volatile("cp.async.bulk.shared::cluster.global"
433 ".mbarrier::complete_tx::bytes [%0], [%1], %2, [%3];"
434 :: "r"(tma_smem(dst)), "l"(tma_gmem(src)), "r"(bytes),
435 "r"(tma_smem(bar)) : "memory");
436}
437
438/* Spin until 'bar' completes the phase of parity 'phase'; the initial parity
439 is 0. try_wait in a C loop rather than a wait loop inside the asm because
440 this is inlined more than once per kernel and PTX labels are not renamed on
441 inlining, so an asm local label would be defined twice.
442
443 The volatile asm and its memory clobber are also what keep the compiler
444 from sinking the wait below the shared reads it guards; the hardware
445 ordering is the mbarrier's own. */
446__device__ __forceinline__ static void tma_wait(unsigned long long *bar,
447 const int phase)
448{
449 unsigned int done;
450
451 do {
452 asm volatile("{\n"
453 ".reg .pred p;\n"
454 "mbarrier.try_wait.parity.shared::cta.b64 p, [%1], %2;\n"
455 "selp.b32 %0, 1, 0, p;\n"
456 "}\n"
457 : "=r"(done) : "r"(tma_smem(bar)), "r"(phase) : "memory");
458 } while (done == 0);
459}

461/* Make this thread's ordinary shared memory writes visible to the async
462 proxy. Reads of shared memory by a bulk copy go through that proxy, and
463 ordinary stores are not ordered before them by __syncthreads() alone, so a
464 store issued without this can read stale shared memory. The load direction
465 needs no counterpart: there the mbarrier completion is what orders the
466 copy's writes before the ordinary reads that consume them. */
467__device__ __forceinline__ static void tma_fence_shared()
468{
469 asm volatile("fence.proxy.async.shared::cta;" ::: "memory");
470}
471
472/* One contiguous shared -> global bulk copy. Tracked by a bulk group rather
473 than an mbarrier: nothing reads the result back, it only has to have been
474 committed before the block releases its shared memory */
475__device__ __forceinline__ static void tma_store(void *dst, const void *src,
476 const int bytes)
477{
478 asm volatile("cp.async.bulk.global.shared::cta.bulk_group [%0], [%1], %2;"
479 :: "l"(tma_gmem(dst)), "r"(tma_smem(src)), "r"(bytes)
480 : "memory");
481}
482
483/* Commit the issued stores and wait for them to have read their source out of
484 shared memory. Only the issuing thread has a group to wait on, but no
485 thread may leave the block before it completes, hence the __syncthreads()
486 that follows this at the call site */
487__device__ __forceinline__ static void tma_store_wait()
488{
489 asm volatile("cp.async.bulk.commit_group;" ::: "memory");
490 asm volatile("cp.async.bulk.wait_group 0;" ::: "memory");
491}
