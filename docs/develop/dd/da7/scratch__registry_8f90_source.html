217 subroutine scratch_registry_init(this, size, expansion_size, dof)
218 class(scratch_registry_t), intent(inout) :: this
219 integer, optional, intent(in) :: size
220 integer, optional, intent(in) :: expansion_size
221 type(dofmap_t), target, intent(in), optional :: dof
222 integer :: initial_size
223
224 call this%free()
225
226 ! Assign the dofmap and expansion size if provided
227 if (present(dof)) this%dof => dof
228 if (present(expansion_size)) this%expansion_size = expansion_size
229
230 ! Allocate the entries and inuse arrays with the initial size
231 initial_size = this%expansion_size
232 if (present(size)) initial_size = size
233
234 if (initial_size .lt. 0) then
235 call neko_error("scratch_registry::init: " // &
236 "Initial size must be non-negative.")
237 end if
238
239 if (this%expansion_size .le. 0) then
240 call neko_error("scratch_registry::init: " // &
241 "Expansion size must be positive.")
242 end if
243
244 allocate(this%entries(initial_size))
245 allocate(this%inuse(initial_size), source = .false.)
246
247 end subroutine scratch_registry_init
248
250 subroutine scratch_registry_free(this)
251 class(scratch_registry_t), intent(inout):: this
252 integer :: i
253
254 if (this%get_n_inuse() .ne. 0) then
255 call neko_error("scratch_registry::free: " // &
256 "Cannot free scratch registry with entries still in use.")
257 end if
258
259 if (allocated(this%entries)) then
260 do i = 1, this%n_entries
261 call this%entries(i)%free()
262 end do
263 deallocate(this%entries)
264 end if
265
266 if (allocated(this%inuse)) deallocate(this%inuse)
267 if (associated(this%dof)) nullify(this%dof)
268
269 ! Reset to default values
270 this%n_entries = 0
271 this%expansion_size = 10
272
273 end subroutine scratch_registry_free

372 subroutine request_host_array(this, scratch_entry, index, n, clear)
373 class(scratch_registry_t), target, intent(inout) :: this
374 real(kind=rp), pointer, dimension(:), intent(inout) :: scratch_entry
375 integer, intent(inout) :: index
376 integer, intent(in) :: n
377 logical, intent(in) :: clear
378 type(host_array_t), pointer :: ptr
379
380 ptr => null()
381
382 ! Look for a compatible, unused object in the registry.
383 !$omp critical (SCRATCH_OMP_LOCK)
384 do index = 1, this%n_entries
385
386 ! Find an unused object of the correct type.
387 if (this%inuse(index)) cycle
388 if (.not. this%entries(index)%is_type('host_array')) cycle
389
390 ! Check compatibility of the object size.
391 ptr => this%entries(index)%get_host_array()
392 if (ptr%size() .eq. n) then
393 this%inuse(index) = .true.
394 exit
395 end if
396 nullify(ptr)
397 end do
398
399 ! If no compatible, unused objects are found, expand and create a new one.
400 if (index .gt. this%n_entries) then
401 this%n_entries = this%n_entries + 1
402
403 if (this%n_entries .gt. this%get_size()) call this%expand()
404 call this%entries(this%n_entries)%init_host_array(n)
405 ptr => this%entries(this%n_entries)%get_host_array()
406
407 this%inuse(this%n_entries) = .true.
408 end if
409 !$omp end critical (SCRATCH_OMP_LOCK)
410
411 ! Assign the pointer to the entry and clear the values if requested.
412 scratch_entry => ptr%x
413 if (clear) call rzero(scratch_entry, n)
414 nullify(ptr)
415
416 end subroutine request_host_array
