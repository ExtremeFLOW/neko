317 !> Set the initial condition of the velocity and pressure. A built-in
318 !! initial condition is set here, a user one by `set_user_initial_condition`.
319 !! @param user Type with user-defined procedures.
320 !! @param is_restart Whether the case is restarting from a checkpoint.
321 subroutine fluid_scheme_base_set_initial_condition(this, user, is_restart)
322 class(fluid_scheme_base_t), intent(inout) :: this
323 type(user_t), intent(in) :: user
324 logical, intent(in) :: is_restart
325 character(len=:), allocatable :: ic_type
326 type(json_file) :: ic_params
327
328 call neko_log%section("Fluid initial condition ")
329
330 if (is_restart) then
331 call neko_log%message("Restart file specified, " // &
332 "initial conditions ignored")
333 else
334 call json_get(this%params, 'case.fluid.initial_condition.type', &
335 ic_type)
336
337 ! Common, non-user initial conditions are handled here by the
338 ! base type.
339 if (trim(ic_type) .ne. 'user') then
340 call json_get(this%params, 'case.fluid.initial_condition', &
341 ic_params)
342 call set_flow_ic_builtin(this%u, this%v, this%w, this%p, this%c_Xh, &
343 this%gs_Xh, ic_type, ic_params)
344 call ic_params%destroy()
345 else
346 call this%set_user_initial_condition(user)
347 end if
348
349 deallocate(ic_type)
350 end if
351
352 call neko_log%end_section()
353
354 end subroutine fluid_scheme_base_set_initial_condition
