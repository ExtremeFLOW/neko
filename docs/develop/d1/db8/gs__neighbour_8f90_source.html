83 subroutine gs_neighbour_init(this, send_pe, recv_pe)
84 class(gs_neighbour_t), intent(inout) :: this
85 type(stack_i4_t), intent(inout) :: send_pe
86 type(stack_i4_t), intent(inout) :: recv_pe
87 integer :: i, nsend, nrecv, send_total, recv_total, ierr
91 integer, allocatable :: src_weights(:), dst_weights(:)
92
93 call this%init_order(send_pe, recv_pe)
94
95 nsend = size(this%send_pe)
96 nrecv = size(this%recv_pe)
97
98 ! Allocate the count/displacement arrays with at least one element so
99 ! that a peerless rank (nsend or nrecv == 0) still passes a valid array
100 ! to the collective.
101 allocate(this%sendcounts(max(1, nsend)), this%sdispls(max(1, nsend)))
102 allocate(this%recvcounts(max(1, nrecv)), this%rdispls(max(1, nrecv)))
103 this%sendcounts = 0
104 this%sdispls = 0
105 this%recvcounts = 0
106 this%rdispls = 0
107
108 send_total = 0
109 do i = 1, nsend
110 this%sendcounts(i) = this%send_dof(this%send_pe(i))%size()
111 this%sdispls(i) = send_total
112 send_total = send_total + this%sendcounts(i)
113 end do
114 allocate(this%send_buf(max(1, send_total)))
115
116 recv_total = 0
117 do i = 1, nrecv
118 this%recvcounts(i) = this%recv_dof(this%recv_pe(i))%size()
119 this%rdispls(i) = recv_total
120 recv_total = recv_total + this%recvcounts(i)
121 end do
122 allocate(this%recv_buf(max(1, recv_total)))
123
124 ! Build a distributed-graph communicator over the halo neighbourhood.
125 ! With MPI_Dist_graph_create_adjacent and reorder = .false. the order of
126 ! sources/destinations is preserved, so the j-th block of the collective
127 ! corresponds to recv_pe(j)/send_pe(j) and the count/displacement arrays
128 ! above can be used directly.
129 allocate(src_weights(max(1, nrecv)), dst_weights(max(1, nsend)))
130 src_weights = 1
131 dst_weights = 1
132 call mpi_dist_graph_create_adjacent(neko_comm, &
133 nrecv, this%recv_pe, src_weights, &
134 nsend, this%send_pe, dst_weights, &
135 mpi_info_null, .false., this%neigh_comm, ierr)
136 deallocate(src_weights, dst_weights)
137
138 end subroutine gs_neighbour_init

162 subroutine gs_nbsend_neighbour(this, u, n, tag, deps, strm)
163 class(gs_neighbour_t), intent(inout) :: this
164 integer, intent(in) :: n
165 real(kind=rp), dimension(n), intent(inout) :: u
166 integer, intent(in) :: tag
167 type(c_ptr), intent(inout) :: deps
168 type(c_ptr), intent(inout) :: strm
169 integer :: i, j, dst, off, ndst, ierr
170
171 ! Gather data from u into the per-peer slab of send_buf according to the
172 ! indices in send_dof. Each thread packs a different peer's slab; the
173 ! implicit barrier at end-do guarantees the buffer is complete before
174 ! the master fires the collective.
175 !$omp do
176 do i = 1, size(this%send_pe)
177 dst = this%send_pe(i)
178 off = this%sdispls(i)
179 ndst = this%sendcounts(i)
180 select type (dofs => this%send_dof(dst)%data)
181 type is (integer)
182 !$omp simd
183 do j = 1, ndst
184 this%send_buf(off + j) = u(dofs(j))
185 end do
186 end select
187 end do
188 !$omp end do
189
190 ! A neighbourhood collective is one call over neigh_comm and must be
191 ! issued by a single thread.
192 !$omp master
193 call mpi_ineighbor_alltoallv(this%send_buf, this%sendcounts, &
194 this%sdispls, mpi_real_precision, this%recv_buf, this%recvcounts, &
195 this%rdispls, mpi_real_precision, this%neigh_comm, this%request, &
196 ierr)
197 !$omp end master
198
199 end subroutine gs_nbsend_neighbour

210 subroutine gs_nbwait_neighbour(this, u, n, op, strm)
211 class(gs_neighbour_t), intent(inout) :: this
212 integer, intent(in) :: n
213 real(kind=rp), dimension(n), intent(inout) :: u
214 type(c_ptr), intent(inout) :: strm
215 integer :: i, j, src, off, nsrc
216 integer :: op
217 integer :: ierr
218
219 ! The collective is master-issued, so wait on it from the master and
220 ! barrier before anyone touches recv_buf.
221 !$omp master
222 call mpi_wait(this%request, mpi_status_ignore, ierr)
223 !$omp end master
224 !$omp barrier
225
226 ! Reduce each received slab into u. The outer loop over peers stays
227 ! serial: a dof shared by 3+ ranks appears in several recv_dof lists, so
228 ! reducing two slabs concurrently would race on that dof. Parallelism is
229 ! taken within each slab instead.
230 do i = 1, size(this%recv_pe)
231 src = this%recv_pe(i)
232 off = this%rdispls(i)
233 nsrc = this%recvcounts(i)
234 select type (dofs => this%recv_dof(src)%data)
235 type is (integer)
236 select case (op)
237 case (gs_op_add)
238 !OCL NORECURRENCE, NOVREC, NOALIAS
239 !DIR$ CONCURRENT
240 !GCC$ ivdep
241 !NEC$ IVDEP
242 !$omp do
243 do j = 1, nsrc
244 u(dofs(j)) = u(dofs(j)) + this%recv_buf(off + j)
245 end do
246 !$omp end do
247 case (gs_op_mul)
248 !OCL NORECURRENCE, NOVREC, NOALIAS
249 !DIR$ CONCURRENT
250 !GCC$ ivdep
251 !NEC$ IVDEP
252 !$omp do
253 do j = 1, nsrc
254 u(dofs(j)) = u(dofs(j)) * this%recv_buf(off + j)
255 end do
256 !$omp end do
257 case (gs_op_min)
258 !OCL NORECURRENCE, NOVREC, NOALIAS
259 !DIR$ CONCURRENT
260 !GCC$ ivdep
261 !NEC$ IVDEP
262 !$omp do
263 do j = 1, nsrc
264 u(dofs(j)) = min(u(dofs(j)), this%recv_buf(off + j))
265 end do
266 !$omp end do
267 case (gs_op_max)
268 !OCL NORECURRENCE, NOVREC, NOALIAS
269 !DIR$ CONCURRENT
270 !GCC$ ivdep
271 !NEC$ IVDEP
272 !$omp do
273 do j = 1, nsrc
274 u(dofs(j)) = max(u(dofs(j)), this%recv_buf(off + j))
275 end do
276 !$omp end do
277 case default
278 call neko_error("Unknown operation in gs_nbwait_neighbour")
279 end select
280 end select
281 end do
282
283 end subroutine gs_nbwait_neighbour
