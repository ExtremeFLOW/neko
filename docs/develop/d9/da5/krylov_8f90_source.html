308 subroutine krylov_init(this, max_iter, rel_tol, abs_tol, M, monitor)
309 class(ksp_t), target, intent(inout) :: this
310 integer, intent(in) :: max_iter
311 real(kind=rp), optional, intent(in) :: rel_tol
312 real(kind=rp), optional, intent(in) :: abs_tol
313 class(pc_t), optional, target, intent(in) :: m
314 logical, optional, intent(in) :: monitor
315
316 call krylov_free(this)
317
318 if (present(rel_tol)) then
319 this%rel_tol = rel_tol
320 else
321 this%rel_tol = ksp_rel_tol
322 end if
323
324 if (present(abs_tol)) then
325 this%abs_tol = abs_tol
326 else
327 this%abs_tol = ksp_abs_tol
328 end if
329
330 this%max_iter = max_iter
331
332 if (present(m)) then
333 this%M => m
334 else
335 if (.not. associated(this%M)) then
336 if (neko_bcknd_device .eq. 1) then
337 allocate(device_ident_t::this%M_ident)
338 else
339 allocate(ident_t::this%M_ident)
340 end if
341 this%M => this%M_ident
342 end if
343 end if
344
345 if (present(monitor)) then
346 this%monitor = monitor
347 else
348 this%monitor = .false.
349 end if
350
351 end subroutine krylov_init
