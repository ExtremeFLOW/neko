88 T rw[LX];

92 const int e = blockIdx.x;

101 shdz[ij_p] = dz[ij];
102 shbr[ij_p] = br[ij];
103 shbs[ij_p] = bs[ij];
104 shbt[ij_p] = bt[ij];

106#pragma unroll

108 ru[k] = u[ij + k*LX*LX + ele];
109 rw[k] = 0.0;

112 __syncthreads();
113#pragma unroll

115 const int ijk = ij + k*LX*LX;
116 const T G00 = g11[ijk+ele];
117 const T G11 = g22[ijk+ele];
118 const T G22 = g33[ijk+ele];
119 const T G01 = g12[ijk+ele];
120 const T G02 = g13[ijk+ele];
121 const T G12 = g23[ijk+ele];
122
123 T ttmp = 0.0;
124 T ttmp_svv = 0.0;
125 shu[ij_p] = ru[k];
126#pragma unroll

128 ttmp += shdz[k+l*P] * ru[l];
129 ttmp_svv += shbt[k+l*P] * ru[l];
130 }
131 __syncthreads();
132
133 T rtmp = 0.0;
134 T stmp = 0.0;
135 T rtmp_svv = 0.0;
136 T stmp_svv = 0.0;
137#pragma unroll
138 for (int l = 0; l < LX; l++) {
139 const T ul_r = shu[l+j*P];
140 const T ul_s = shu[i+l*P];
141 rtmp += shdx[i+l*P] * ul_r;
142 stmp += shdy[j+l*P] * ul_s;
143 rtmp_svv += shbr[i+l*P] * ul_r;
144 stmp_svv += shbs[j+l*P] * ul_s;
145 }
146
147 const T H1 = h1[ijk+ele];
148 shur[ij] = H1 * (G00 * rtmp + G01 * stmp + G02 * ttmp);
149 shus[ij_p] = H1 * (G01 * rtmp + G11 * stmp + G12 * ttmp);
150 rut = H1 * (G02 * rtmp + G12 * stmp + G22 * ttmp);
151
152 const T H1_SVV = h1_svv[ijk+ele];
153 shur_svv[ij] = H1_SVV
154 * (G00 * rtmp_svv + G01 * stmp_svv + G02 * ttmp_svv);
155 shus_svv[ij_p] = H1_SVV
156 * (G01 * rtmp_svv + G11 * stmp_svv + G12 * ttmp_svv);
157 rut_svv = H1_SVV
158 * (G02 * rtmp_svv + G12 * stmp_svv + G22 * ttmp_svv);
159
160 __syncthreads();
161
162 T wijke = 0.0;
163#pragma unroll
164 for (int l = 0; l < LX; l++) {
165 wijke += shur[l+j*LX] * shdx[l+i*P]
166 + shur_svv[l+j*LX] * shbr[l+i*P];
167 wijke += shus[i+l*P] * shdy[l+j*P]
168 + shus_svv[i+l*P] * shbs[l+j*P];
169 rw[l] += rut * shdz[k+l*P] + rut_svv * shbt[k+l*P];
170 }
171 rw[k] += wijke;
172 }
173
174#pragma unroll
175 for (int k = 0; k < LX; ++k) {
176 w[ij + k*LX*LX + ele] = rw[k];
177 }
178}
