46void opencl_buffer_reserve(opencl_buffer_t *buf, size_t size) {
47 cl_int err;
48
49 if (size <= buf->size)
50 return;
51
52 if (buf->host != NULL) {
53 free(buf->host);
54 CL_CHECK(clReleaseMemObject(buf->dev));
55 }
56
57 buf->host = malloc(size);
58 buf->dev = clCreateBuffer(glb_ctx, CL_MEM_READ_WRITE, size, NULL, &err);
59 CL_CHECK(err);
60 buf->size = size;
61
62 if (!buf->registered) {
63 buf->next = buffer_registry;
64 buffer_registry = buf;
65 buf->registered = 1;
66 }
67}

69void opencl_buffer_free_all(void) {
70 opencl_buffer_t *buf = buffer_registry;
71
72 while (buf != NULL) {
73 opencl_buffer_t *next = buf->next;
74 if (buf->host != NULL) {
75 free(buf->host);
76 buf->host = NULL;
77 CL_CHECK(clReleaseMemObject(buf->dev));
78 buf->dev = NULL;
79 }
80 buf->size = 0;
81 buf->registered = 0;
82 buf->next = NULL;
83 buf = next;
84 }
85 buffer_registry = NULL;
86}
