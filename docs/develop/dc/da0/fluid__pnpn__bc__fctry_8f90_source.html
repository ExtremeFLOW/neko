144 module subroutine fluid_pnpn_bc_split(json, velocity_json, pressure_json, &
145 has_velocity, has_pressure)
146 type(json_file), intent(inout) :: json
147 type(json_file), intent(inout) :: velocity_json
148 type(json_file), intent(inout) :: pressure_json
149 logical, intent(out) :: has_velocity
150 logical, intent(out) :: has_pressure
151 character(len=:), allocatable :: type
152 character(len=25) :: velocity_type, pressure_type
153 logical :: couple_pressure
154 integer :: i
155
156 velocity_type = ""
157 pressure_type = ""
158 has_velocity = json%valid_path("velocity")
159 has_pressure = json%valid_path("pressure")
160
161 if (json%valid_path("type")) then
162 if (has_velocity .or. has_pressure) then
163 call neko_error("A fluid boundary condition cannot have both a " // &
164 "'type' keyword and a 'velocity' or 'pressure' object.")
165 end if
166
167 call json_get(json, "type", type)
168
169 do i = 1, size(fluid_pnpn_bc_presets)
170 if (trim(type) .eq. trim(fluid_pnpn_bc_presets(i)%name)) exit
171 end do
172
173 if (i .gt. size(fluid_pnpn_bc_presets)) then
174 call neko_type_error("fluid_pnpn boundary conditions", type, &
175 FLUID_PNPN_BC_PRESETS%name)
176 end if
177
178 velocity_type = fluid_pnpn_bc_presets(i)%velocity
179 pressure_type = fluid_pnpn_bc_presets(i)%pressure
180
181 ! The overset interface only constrains the pressure on request.
182 if (trim(type) .eq. "overset_interface") then
183 call json_get_or_default(json, "couple_pressure", couple_pressure, &
184 .false.)
185 if (.not. couple_pressure) pressure_type = ""
186 end if
187
188 has_velocity = len_trim(velocity_type) .gt. 0
189 has_pressure = len_trim(pressure_type) .gt. 0
190 else if (.not. (has_velocity .and. has_pressure)) then
191 call neko_error("A fluid boundary condition needs either a 'type' " // &
192 "keyword, or both a 'velocity' and a 'pressure' object.")
193 end if
194
195 if (has_velocity) then
196 call fluid_pnpn_bc_extract_part(json, "velocity", velocity_type, &
197 velocity_json)
198 end if
199
200 if (has_pressure) then
201 call fluid_pnpn_bc_extract_part(json, "pressure", pressure_type, &
202 pressure_json)
203 end if
204 end subroutine fluid_pnpn_bc_split

216 subroutine fluid_pnpn_bc_extract_part(entry, key, preset_type, part)
217 type(json_file), intent(inout) :: entry
218 character(len=*), intent(in) :: key
219 character(len=*), intent(in) :: preset_type
220 type(json_file), intent(inout) :: part
221 type(json_core) :: core
222 type(json_value), pointer :: root, sub, child, existing, child_clone
223 character(len=:), allocatable :: buffer, child_name
224 logical :: found
225 integer :: i, n
226
227 ! Start from an independent copy of the entry
228 call entry%print_to_string(buffer)
229 call part%destroy()
230 call part%initialize(strict_type_checking = .true.)
231 call part%load_from_string(buffer)
232
233 ! Retain only the top-level stuff like the zone indices
234 call part%remove("type")
235 call part%remove("velocity")
236 call part%remove("pressure")
237
238 if (len_trim(preset_type) .gt. 0) then
239 call part%add("type", trim(preset_type))
240 end if
241
242 ! The sub-object is read from the unmodified entry, not from the copy.
243 call entry%get(key, sub, found)
244 if (found) then
245 call part%get_core(core)
246 call part%get(root)
247 n = core%count(sub)
248 do i = 1, n
249 call core%get_child(sub, i, child, found)
250 call core%info(child, name = child_name)
251
252 call core%get_child(root, child_name, existing, found)
253 if (found) call core%remove(existing, destroy = .true.)
254
255 nullify(child_clone)
256 call core%clone(child, child_clone)
257 call core%add(root, child_clone)
258 end do
259 end if
260 end subroutine fluid_pnpn_bc_extract_part

401 subroutine fluid_pnpn_bc_register(registry, registry_size, known_bcs, &
402 base_type, type_name, allocator)
403 type(fluid_pnpn_bc_allocator_entry), allocatable, intent(inout) :: &
404 registry(:)
405 integer, intent(inout) :: registry_size
406 character(len=*), intent(in) :: known_bcs(:)
407 character(len=*), intent(in) :: base_type
408 character(len=*), intent(in) :: type_name
409 procedure(fluid_pnpn_bc_allocate), pointer, intent(in) :: allocator
410 type(fluid_pnpn_bc_allocator_entry), allocatable :: temp(:)
411 integer :: i
412
413 do i = 1, size(known_bcs)
414 if (trim(type_name) .eq. trim(known_bcs(i))) then
415 call neko_type_registration_error(base_type, type_name, .true.)
416 end if
417 end do
418
419 do i = 1, registry_size
420 if (trim(type_name) .eq. trim(registry(i)%type_name)) then
421 call neko_type_registration_error(base_type, type_name, .false.)
422 end if
423 end do
424
425 if (registry_size .eq. 0) then
426 allocate(registry(1))
427 else
428 allocate(temp(registry_size + 1))
429 temp(1:registry_size) = registry
430 call move_alloc(temp, registry)
431 end if
432
433 registry_size = registry_size + 1
434 registry(registry_size)%type_name = type_name
435 registry(registry_size)%allocator => allocator
436 end subroutine fluid_pnpn_bc_register

450 module subroutine pressure_bc_factory(object, scheme, json, coef, user)
451 class(bc_t), pointer, intent(inout) :: object
452 type(fluid_pnpn_t), intent(in) :: scheme
453 type(json_file), intent(inout) :: json
454 type(coef_t), target, intent(in) :: coef
455 type(user_t), target, intent(in) :: user
456 character(len=:), allocatable :: type
457 integer :: i
458 integer, allocatable :: zone_indices(:)
459 character(len=:), allocatable :: default_name
460 character(len=64) :: buf
461
462 call json_get(json, "type", type)
463
464 if (trim(type) .eq. "neumann") then
465 ! The flux is set by the scheme and is not a user input.
466 if (json%valid_path("flux")) then
467 call neko_error("The 'neumann' pressure boundary condition " // &
468 "does not accept a 'flux' keyword, the flux is set by the " // &
469 "scheme.")
470 end if
471
472 if (associated(object)) then
473 call object%free()
474 deallocate(object)
475 end if
476 return
477 end if
478
479 call fluid_pnpn_pressure_bc_allocator(object, type)
480
481 select case (trim(type))
482 case ("user_dirichlet")
483 select type (obj => object)
484 type is (field_dirichlet_t)
485 obj%update => user%dirichlet_conditions
486 call json%add("field_name", scheme%p%name)
487 end select
488 case ("overset_interface")
489 select type (obj => object)
490 type is (overset_interface_t)
491 call json%add("field_name", scheme%p%name)
492 obj%morph_interface => user%morph_interface
493 end select
494 end select
495
496 call json_get_or_lookup(json, "zone_indices", zone_indices)
497 call object%init(coef, json)
498
499 do i = 1, size(zone_indices)
500 call object%mark_labeled_zone(zone_indices(i))
501 end do
502
503 write(buf, '("pressure_bc_", I0)') zone_indices(1)
504 default_name = trim(buf)
505 call json_get_or_default(json, "name", object%name, default_name)
506 object%zone_indices = zone_indices
507 call object%finalize()
508
509 deallocate(type)
510 deallocate(zone_indices)
511 deallocate(default_name)
512 end subroutine pressure_bc_factory

527 module subroutine velocity_bc_factory(object, scheme, json, coef, user)
528 class(bc_t), pointer, intent(inout) :: object
529 type(fluid_pnpn_t), intent(in) :: scheme
530 type(json_file), intent(inout) :: json
531 type(coef_t), target, intent(in) :: coef
532 type(user_t), target, intent(in) :: user
533 character(len=:), allocatable :: type
534 integer :: i
535 integer, allocatable :: zone_indices(:)
536 character(len=:), allocatable :: default_name
537 character(len=:), allocatable :: bc_name
538 character(len=64) :: buf
539 integer :: var_type, n_children
540
541 call json_get(json, "type", type)
542
543 if (trim(type) .eq. "neumann") then
544 if (.not. json%valid_path("flux")) then
545 if (associated(object)) then
546 call object%free()
547 deallocate(object)
548 end if
549 return
550 end if
551
552 ! neumann_t applies the flux component-wise to a vector, so a scalar
553 ! flux is not enough.
554 call json%info("flux", var_type = var_type, n_children = n_children)
555 if (var_type .ne. json_array .or. n_children .ne. 3) then
556 call neko_error("The 'flux' of the 'neumann' velocity boundary " // &
557 "condition must be an array of 3 reals.")
558 end if
559 end if
560
561 call fluid_pnpn_velocity_bc_allocator(object, type, &
562 scheme%full_stress_formulation)
563
564 select case (trim(type))
565 case ("wall_model")
566 ! Kind of hack, but OK for now
567 call json%add("scheme_name", scheme%name)
568
569 select type (wall_bc => object)
570 type is (wall_model_bc_t)
571 wall_bc%user => user
572 end select
573 case ("user_dirichlet")
574 select type (obj => object)
575 type is (field_dirichlet_vector_t)
576 obj%update => user%dirichlet_conditions
577 end select
578 case ("overset_interface")
579 select type (obj => object)
580 type is (overset_interface_vector_t)
581 obj%morph_interface => user%morph_interface
582 end select
583 end select
584
585 call json_get_or_lookup(json, "zone_indices", zone_indices)
586 write(buf,'("velocity_bc_",I0)') zone_indices(1)
587 default_name = trim(buf)
588 call json_get_or_default(json, "name", bc_name, default_name)
589
590 call object%init(coef, json)
591 do i = 1, size(zone_indices)
592 call object%mark_labeled_zone(zone_indices(i))
593 end do
594
595 object%name = bc_name
596 object%zone_indices = zone_indices
597
598 call object%finalize()
599
600 deallocate(type)
601 deallocate(zone_indices)
602 deallocate(default_name)
603 deallocate(bc_name)
604 end subroutine velocity_bc_factory
